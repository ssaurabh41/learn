# examples

A small placeholder design and its configs, used by the regression test.
It follows the usual layout: `$PROJ_ROOT/<block>/rtl/source.f` and
`$PROJ_ROOT/<block>/syn/outputs/<top>.syn.v`, with `PROJ_ROOT` set to this
directory. The files only need to exist for flowgen: they are not a real
library or a real synthesis result, so Formality itself will not pass on
them.

```
configs/project.config.tcl   project settings (library, tool setup)
configs/block.config.tcl     block config; uses the search dirs
block/rtl/                   RTL, source.f, include/
block/syn/outputs/           netlist and SVF
pdk/stdcell_tt.db            placeholder library
env/setup.cshrc              placeholder tool setup
```

Run the regression after every change to flowgen:

```
examples/run_regression.csh
```

It generates run directories under `regression_out/` (next to `examples/`, so
unpacking a new bundle leaves it alone) and checks
that flowgen prints no warnings and that the run directories are complete.

#!/bin/tcsh -f
# Regression test for flowgen. Usage: examples/run_regression.csh
# Needs python3 (3.6+ with tkinter) and tcsh on PATH.
#
# Each case generates a run directory in regression_out/<case>/ at the top
# (removed at the start) and must print no warnings or errors:
#   example   examples/configs/block.config.tcl (uses the search dirs)
#   new_block flowgen/configs/new_block.config.tcl, as --new writes it
#   csv       flowgen/configs/block_template.config.csv through csv2config.py
# The run directory must hold the run script (which must parse), the
# Formality script, logs/, reports/ and copies of both configs.
# Last, the reference template block_template.config.tcl must be valid
# (its paths do not exist here, so it is run with --force).

set here = "`dirname $0`"
set root = "`cd $here/.. && pwd`"
if ( "$root" == "" || ! -f "$root/flowgen/flowgen.py" ) then
  echo "regression: cannot find the repo root from $0"
  exit 2
endif
set flowgen = "$root/flowgen/flowgen.py"
set out = "$root/regression_out"   # outside examples/: unpack.py leaves it alone
if ( -d "$out" ) rm -rf "$out"
mkdir -p "$out/configs"
set fail = 0

# Configs for the cases. The copies in regression_out/configs/ get the
# example project config with PROJ_ROOT as a full path.
set q = '"'
sed "s|^set PROJ_ROOT .*|set PROJ_ROOT ${q}$root/examples${q}|" \
  "$root/examples/configs/project.config.tcl" > "$out/configs/project.config.tcl"
sed "s|@BLOCK@|block|g" "$root/flowgen/configs/new_block.config.tcl" \
  > "$out/configs/new_block.config.tcl"
python3 "$root/flowgen/csv2config.py" "$root/flowgen/configs/block_template.config.csv" \
  -o "$out/configs/csv.config.tcl" > "$out/csv2config.log"
if ( $status != 0 ) then
  echo "FAIL csv: csv2config failed (see $out/csv2config.log)"
  set fail = 1
endif

foreach case ( example new_block csv )
  if ( "$case" == "example" ) then
    set config = "$root/examples/configs/block.config.tcl"
  else
    set config = "$out/configs/$case.config.tcl"
    if ( ! -f "$config" ) continue
  endif
  mkdir -p "$out/$case"
  cd "$out/$case"
  python3 "$flowgen" "$config" >& flowgen.log
  if ( $status != 0 ) then
    echo "FAIL ${case}: flowgen failed (see $out/$case/flowgen.log)"
    set fail = 1
    continue
  endif
  grep -q -e "WARNING" -e "ERROR" flowgen.log
  if ( $status == 0 ) then
    echo "FAIL ${case}: flowgen printed warnings (see $out/$case/flowgen.log)"
    set fail = 1
    continue
  endif
  set rundir = "`ls -d $out/$case/*_block_formality`"
  set casefail = 0
  foreach f ( run_rtl2syn.csh rtl2syn/rtl2syn.tcl rtl2syn/logs rtl2syn/reports \
              project.config.tcl `basename $config` )
    if ( ! -e "$rundir/$f" ) then
      echo "FAIL ${case}: missing $f in the run directory"
      set casefail = 1
    endif
  end
  tcsh -n "$rundir/run_rtl2syn.csh"
  if ( $status != 0 ) then
    echo "FAIL ${case}: run_rtl2syn.csh does not parse"
    set casefail = 1
  endif
  if ( $casefail == 0 ) then
    echo "PASS $case"
  else
    set fail = 1
  endif
end

# The reference template must be a valid config.
mkdir -p "$out/template"
cd "$out/template"
python3 "$flowgen" --force "$root/flowgen/configs/block_template.config.tcl" >& flowgen.log
if ( $status != 0 ) then
  echo "FAIL template: block_template.config.tcl is not a valid config (see $out/template/flowgen.log)"
  set fail = 1
else
  echo "PASS template"
endif

if ( $fail != 0 ) then
  echo "regression: FAILED"
  exit 1
endif
echo "regression: PASSED"
exit 0
#@@END examples/run_regression.csh

# flowgen

Generates EDA run directories from one Tcl block config. Python 3.6+ (with tkinter), standard library only.

The first supported tool is Synopsys Formality, with the `rtl2syn` flow: RTL reference vs. synthesized netlist.

## Quick start

```
python3 unpack.py bundle.txt            once: creates flowgen/
flowgen/runflowgen.sh --new uart_ctrl   creates flowgen/configs/uart_ctrl.config.tcl
flowgen/runflowgen.sh uart_ctrl         creates the run directory
<run directory>/run_rtl2syn.csh         runs Formality, prints PASS or FAIL
```

Each command prints what to do next. `uart_ctrl` is your block's name.

1. `--new` creates a short block config, `flowgen/configs/uart_ctrl.config.tcl`. The first time it also creates `flowgen/configs/project.config.tcl`, the settings shared by all blocks.
2. Edit `project.config.tcl` once per project: `PROJ_ROOT` and `LIB_FILES`, plus the tool setup (`ENV_SOURCE`, `ENV_SETENV`) if you need it.
3. Check the block config: `TOP_DESIGN` and the paths to the RTL file list, the netlist and the SVF. They are built from `$PROJ_ROOT/$BLOCK`, so a block that follows the usual layout needs no change.
4. Generate. If a file is missing, the error list says which setting to fix.

Your configs live in `flowgen/configs/`, next to the templates. Keep a copy: updating flowgen replaces `flowgen/` (see Updating flowgen below). The commands can be run from any directory; the run directory is created in the current directory.

### Or: fill in a spreadsheet

Instead of editing the Tcl config, you can fill in a spreadsheet and convert it:

```
cp flowgen/configs/block_template.config.csv flowgen/configs/uart_ctrl.config.csv
```

Open `flowgen/configs/uart_ctrl.config.csv` in LibreOffice (in the import dialog set the Value column to Text), fill in the Value column and save it as CSV. Every setting has a Description and an Example column. One value per row; for a list, add more rows with the same Setting (or an empty Setting cell). An empty Value means "not set here", so the project config's value is used. Then:

```
flowgen/csv2config.py flowgen/configs/uart_ctrl.config.csv
flowgen/runflowgen.sh uart_ctrl
```

`csv2config.py` writes `flowgen/configs/uart_ctrl.config.tcl`. It does not overwrite an existing file unless you add `--force`; once you use the CSV, edit the CSV and rerun it rather than editing the `.tcl`.

## Usage

```
runflowgen.sh --new BLOCK
runflowgen.sh [--force] BLOCK
```

`runflowgen.sh` is a csh (tcsh) wrapper that runs `flowgen.py` with `python3`, passing all arguments through. `BLOCK` reads `flowgen/configs/BLOCK.config.tcl`; a path to a config file also works. A missing input file (RTL, netlist, library, SVF, hook, cshrc) is an error. Add `--force` to generate anyway, for example before all design files exist: the missing files become warnings and are listed at the top of the generated scripts.

This creates the following in the current directory. An existing directory is never overwritten: a second run in the same second gets a `_2` suffix.

```
<yyyy_mm_dd_hh_mm_ss>_<TOP_DESIGN>_formality/
  uart_ctrl.config.tcl  copy of the block config used
  project.config.tcl    copy of the project config used
  run_rtl2syn.csh       run script (tcsh): sets up the env, runs fm_shell
  rtl2syn/              where Formality runs
    rtl2syn.tcl         generated Formality script
    logs/               empty, filled in by the tool
    reports/            empty, filled in by the tool
```

Nothing is launched. To run a flow:

```
cd <yyyy_mm_dd_hh_mm_ss>_<TOP_DESIGN>_formality
./run_rtl2syn.csh
```

The log goes to `rtl2syn/logs/rtl2syn.log`, the reports to `rtl2syn/reports/`, and the result (PASS or FAIL) to `rtl2syn/rtl2syn.status`. The run script exits with 0 for PASS, 1 for FAIL, and 2 if the flow did not finish (a setup file is missing, the tool did not start, or the script stopped early).

## Config

The config is plain Tcl. A block config starts with `source project.config.tcl`, so every setting can go in the project config (all blocks) or in the block config (one block); a block config can set a project setting again to change it. `configs/block_template.config.tcl` is the reference: it explains every setting with examples. Copy the settings you need from it.

- Single values go in double quotes: `set TOP_DESIGN "block"`.
- Lists go in braces, one double-quoted item per line. flowgen expands `$VARIABLES` inside list items.
- Relative paths are resolved from the config file's directory (`flowgen/configs/`).
- File lists (`RTL_FILES`, `SYN_FILES`, `APR_FILES`, `SIM_FILES`) accept `.f` files. A `.f` file holds one path per line, with `//` or `#` comments (at the start of a line or after a space). It may also contain `-f`, `-v`, `+incdir+` and `+define+`. `$VAR` in a `.f` file is a config variable (e.g. your own `USER_RTL_DIR`), else an environment variable; it must be set when flowgen runs. A file listed more than once is read once.
- A file name without a full path (in `RTL_FILES`, `SYN_FILES` or a `.f` file) is looked up next to the config or its `.f` file first, then in `RTL_SEARCH_DIRS` / `SYN_SEARCH_DIRS` in order, then (RTL only) in `RTL_INCDIRS` and `+incdir+` dirs. A name found in several places uses the first, with a warning; a name found nowhere is an error that lists the directories searched.
- A name that looks like a misspelled setting (`BLACK_BOX_MODULES`, `top_design`) produces a warning. A tool name other than `formality` in `TOOL_CMD`, `TOOL_SETTINGS` or `TOOL_HOOKS` is an error.
- Errors are all reported together, and nothing is generated if there are any.

Regression: `examples/run_regression.csh` runs flowgen on the example design in `examples/` and on the `--new` template, and checks that nothing is missing and no warning is printed. Run it after unpacking a new bundle.

Updating flowgen: unpack the new bundle with `python3 unpack.py --force bundle.txt`. It replaces `flowgen/`. If `flowgen/` holds files of your own, such as your configs in `flowgen/configs/` or run directories, the old directory is kept as `flowgen.old_<date>_<time>` instead of being deleted. Copy your configs back: `cp flowgen.old_<date>_<time>/configs/*.config.* flowgen/configs/` (the shipped templates are named `*_template.*` and `new_block.config.tcl`; do not copy those back).

#!/usr/bin/env python3
"""csv2config - turn a block config spreadsheet (CSV) into a block config.

Usage:
    csv2config.py [--force] [-o OUTPUT] CSV

Arguments:
    CSV       The block config spreadsheet, <block>.config.csv, normally in
              flowgen/configs/. Start from configs/block_template.config.csv:
                cp flowgen/configs/block_template.config.csv \\
                   flowgen/configs/uart_ctrl.config.csv
              Edit it in LibreOffice (when opening, set the Value column to
              Text) and save it as CSV.
    -o        Output file. Default: the CSV name with .csv replaced by
              .tcl, next to the CSV (uart_ctrl.config.csv ->
              uart_ctrl.config.tcl), which is where runflowgen.sh finds it.
    --force   Overwrite the output file if it already exists.
    -h        Show this help and exit.

Example:
    csv2config.py flowgen/configs/uart_ctrl.config.csv
    runflowgen.sh uart_ctrl

CSV columns: Setting, Value, Required, Description, Example.
  - One value per row. For a list, add more rows with the same Setting
    (or an empty Setting cell); they are collected in order.
  - A Setting starting with # is a heading.
  - An empty Value means "not set here", so the project config's value
    (or flowgen's default) is used. Empty rows are left out of the output.
  - Setting "source" writes "source <file>", e.g. project.config.tcl.
  - Description becomes a comment above the setting. Required and Example
    are for the reader only.
  - Settings flowgen does not know are written as your own helper
    variables (e.g. USER_RTL_DIR), so you can use them as $NAME.

Python 3.6+, standard library only.
"""

import argparse
import csv
import os
import re
import sys
import textwrap

sys.dont_write_bytecode = True        # no __pycache__/ in flowgen/ (unpack.py
                                      # would treat it as a file of yours)
from flowgen import LIST_SETTINGS     # flowgen.py in the same directory

NAME_RE = re.compile(r"^[A-Za-z_][A-Za-z0-9_]*(\([^()\s]+\))?\Z")
COLUMNS = ("setting", "value", "required", "description", "example")


class CsvError(Exception):
    pass


def tcl_quote(text):
    """Text in double quotes for Tcl. $NAME stays live (variables expand);
    characters Tcl would treat specially otherwise are escaped."""
    for ch in ("\\", '"', "[", "]"):
        text = text.replace(ch, "\\" + ch)
    return '"%s"' % text


def read_rows(path):
    """Rows as dicts with the COLUMNS keys, plus the CSV line number."""
    try:
        with open(path, "rb") as fh:
            raw = fh.read()
    except (IOError, OSError) as exc:
        raise CsvError("cannot read %s: %s" % (path, exc))
    try:
        text = raw.decode("utf-8-sig")
    except UnicodeDecodeError:
        raise CsvError("%s is not UTF-8/ASCII text; save it as CSV with UTF-8" % path)
    lines = text.splitlines(True)
    if not lines:
        raise CsvError("%s is empty" % path)
    # LibreOffice may save with ; or tab instead of , depending on settings.
    delimiter = max((",", ";", "\t"), key=lines[0].count)
    reader = csv.reader(lines, delimiter=delimiter)
    head = [h.strip().lower() for h in next(reader)]
    if head[:2] != ["setting", "value"]:
        raise CsvError("%s: first row must be the header Setting, Value, ... (got %s)"
                       % (path, ", ".join(head)))
    rows = []
    for lineno, cells in enumerate(reader, 2):
        cells = [c.strip() for c in cells] + [""] * len(COLUMNS)
        row = dict(zip(COLUMNS, cells))
        row["line"] = lineno
        rows.append(row)
    return rows


def collect(rows):
    """Group rows into ordered entries: ['heading', title] or
    ['setting', name, values, description, first line]."""
    entries, by_name, current, errors = [], {}, None, []
    for row in rows:
        name, value = row["setting"], row["value"]
        if not any(row[c] for c in COLUMNS):
            continue
        if name.startswith("#"):
            entries.append(["heading", name.lstrip("#").strip()])
            current = None
            continue
        if not name:
            if not value:
                continue
            if current is None:
                errors.append("line %d: value %r has no Setting above it" % (row["line"], value))
                continue
            name = current
        if name != "source" and not NAME_RE.match(name):
            errors.append("line %d: %r is not a valid setting name" % (row["line"], name))
            continue
        if name in by_name and name not in LIST_SETTINGS and name != "source":
            errors.append("line %d: %s is a single value but appears again (first on line %d)"
                          % (row["line"], name, by_name[name][4]))
            continue
        if name in by_name and name != "source":
            entry = by_name[name]
        else:
            entry = ["setting", name, [], row["description"], row["line"]]
            by_name[name] = entry
            entries.append(entry)
        if value:
            entry[2].append(value)
        current = name
    if errors:
        raise CsvError("\n".join(errors))
    return entries


def render(entries, csv_name):
    """Config text. Empty settings are left out; a heading is written only
    if a setting follows it."""
    blocks, heading = [], None
    for entry in entries:
        if entry[0] == "heading":
            heading = entry[1]
            continue
        _, name, values, description, _ = entry
        if not values:
            continue
        lines = []
        if heading is not None:
            lines += ["", "# " + heading]
            heading = None
        lines.append("")
        for para in description.splitlines():
            lines += ["# " + l for l in textwrap.wrap(para, 74)]
        if name == "source":
            lines += ["source %s" % tcl_quote(v) for v in values]
        elif name in LIST_SETTINGS:
            lines.append("set %s {" % name)
            lines += ["  " + tcl_quote(v) for v in values]
            lines.append("}")
        else:
            lines.append("set %s %s" % (name, tcl_quote(values[0])))
        blocks += lines
    header = "# From %s. Edit the CSV and rerun csv2config.py." % csv_name
    return "\n".join([header] + blocks) + "\n"


def main(argv=None):
    parser = argparse.ArgumentParser(
        description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter)
    parser.add_argument("csv", metavar="CSV", help="block config CSV, <block>.config.csv")
    parser.add_argument("-o", "--output", help="output .config.tcl file")
    parser.add_argument("--force", action="store_true", help="overwrite an existing output file")
    args = parser.parse_args(argv)

    if not args.csv.endswith(".config.csv"):
        print("csv2config: WARNING: block CSVs are named <block>.config.csv: %s" % args.csv)
    output = args.output or (args.csv[:-len(".csv")] + ".tcl" if args.csv.endswith(".csv")
                             else args.csv + ".tcl")
    if os.path.lexists(output) and not args.force:
        sys.exit("csv2config: ERROR: %s already exists. Use --force to overwrite it." % output)

    try:
        entries = collect(read_rows(args.csv))
    except CsvError as exc:
        sys.exit("\n".join("csv2config: ERROR: " + l for l in str(exc).splitlines()))
    text = render(entries, os.path.basename(args.csv))
    bad = sorted(set(ch for ch in text if ord(ch) > 126))
    if bad:
        sys.exit("csv2config: ERROR: the CSV has non-ASCII characters (%s); values must be "
                 "plain ASCII" % " ".join(ascii(ch) for ch in bad))

    with open(output, "w", encoding="ascii", newline="\n") as fh:
        fh.write(text)
    print("csv2config: wrote %s" % output)
    block = os.path.basename(output)
    if block.endswith(".config.tcl"):
        print("csv2config: next, generate the run directory: runflowgen.sh %s"
              % block[:-len(".config.tcl")])
    return 0


if __name__ == "__main__":
    sys.exit(main())

def load_tcl_config(path):
    """Source the config in a fresh Tcl interpreter.

    Returns ({name: str or {key: str}}, interpreter) for the global
    variables the config created (arrays become dicts; ENV_SETENV keeps
    the order in which the config set its entries).
    """
    tcl = tkinter.Tcl()
    before = set(tcl.splitlist(tcl.eval("info globals")))
    # Remember the order in which ENV_SETENV(...) entries are set.
    tcl.eval("namespace eval ::flowgen {variable env_order {}}")
    tcl.eval("trace add variable ::ENV_SETENV write "
             "{apply {{name key op} {lappend ::flowgen::env_order $key}}}")

    old_cwd = os.getcwd()
    os.chdir(os.path.dirname(path))      # relative paths/globs resolve here
    try:
        tcl.call("source", path)
    except tkinter.TclError as exc:
        detail = tcl.eval("set ::errorInfo")
        raise ConfigError("Tcl error while reading %s: %s\n%s" % (path, exc, detail))
    finally:
        os.chdir(old_cwd)

    values = {}
    for name in sorted(set(tcl.splitlist(tcl.eval("info globals"))) - before):
        if not IDENT_RE.match(name):
            continue  # flowgen variables are always plain identifiers
        if tcl.eval("array exists ::%s" % name) == "1":
            flat = tcl.splitlist(tcl.eval("array get ::%s" % name))
            values[name] = dict(zip(flat[0::2], flat[1::2]))
        elif tcl.eval("info exists ::%s" % name) == "1":
            values[name] = tcl.eval("set ::%s" % name)
    env = values.get("ENV_SETENV")
    if isinstance(env, dict):
        ordered = collections.OrderedDict()
        for key in tcl.splitlist(tcl.eval("set ::flowgen::env_order")):
            if key in env and key not in ordered:
                ordered[key] = env[key]
        for key in sorted(env):
            ordered.setdefault(key, env[key])
        values["ENV_SETENV"] = ordered
    return values, tcl

            items.append(item)


class Config(object):
    """Typed, validated view of the block config. Collects all problems."""

    def __init__(self, path, force=False):
        self.force = force
        self.forced = []      # file errors downgraded to warnings by --force
        self.path = os.path.realpath(path)   # follow links
        self.dir = os.path.dirname(self.path)
        self.errors = []
        self.warnings = []
        self._raw, self._tcl = load_tcl_config(self.path)

        self.file_problems = 0    # missing or unreadable input files seen

        for name in sorted(self._raw):
            if name in KNOWN_VARS:
                continue
            close = difflib.get_close_matches(name.upper(), KNOWN_VARS, n=1, cutoff=0.75)
            if close:
                self.warnings.append("unknown variable '%s' ignored (did you mean %s?)"
                                     % (name, close[0]))

        self.top = self.scalar("TOP_DESIGN", required=True)
        if self.top and not IDENT_RE.match(self.top):
            self.errors.append("TOP_DESIGN: \"%s\" is not a module name (letters, digits "
                               "and _ only)" % self.top)
        self.flows = []
        add_unique(self.flows, self.tcl_list("FLOWS", default=["rtl2syn"]))
        for flow in self.flows:
            if flow not in SUPPORTED_FLOWS:
                self.errors.append("FLOWS: unknown flow '%s' (supported: %s)"
                                   % (flow, ", ".join(SUPPORTED_FLOWS)))

        # Design data
        self.rtl_format = self.choice("RTL_FORMAT", RTL_READERS, default="sverilog")
        self.rtl = self.fileset("RTL_FILES", "RTL_SEARCH_DIRS", required=True,
                                incdirs=self.paths("RTL_INCDIRS"))
        add_unique(self.rtl.defines, self.tcl_list("RTL_DEFINES"))
        for item in self.rtl.incdirs + self.rtl.defines:
            if re.search(r"\s", item):
                self.errors.append("RTL_INCDIRS / RTL_DEFINES: spaces are not supported "
                                   "in include directories and defines: \"%s\"" % item)
        if self.rtl_format == "vhdl" and (self.rtl.incdirs or self.rtl.defines):
            self.errors.append("RTL_INCDIRS / RTL_DEFINES (or +incdir+/+define+ in a .f "
                               "file) only apply to verilog/sverilog")
        needs_syn = "rtl2syn" in self.flows
        self.syn = self.fileset("SYN_FILES", "SYN_SEARCH_DIRS", required=needs_syn, netlist=True)
        # APR_FILES, SDC_FILES and UPF_FILES are not read by any flow yet,
        # so their files are not checked.
        self.apr = self.tcl_list("APR_FILES")
        svf = self.scalar("SVF_FILE")
        self.svf = self.resolve("SVF_FILE", svf) if svf else ""
        self.sdc = self.tcl_list("SDC_FILES")
        self.upf = self.tcl_list("UPF_FILES")
        if self.upf:
            self.warnings.append("UPF_FILES is set but not used yet: the generated "
                                 "Formality script does not load UPF")

        # Libraries and simulation models
        self.lib_format = self.choice("LIB_FORMAT", LIB_READERS, default="db")
        self.libs = self.paths("LIB_FILES", required=True)
        if self.lib_format in LIB_EXTENSIONS:
            for lib in self.libs:
                if not lib.endswith(LIB_EXTENSIONS[self.lib_format]):
                    self.warnings.append("LIB_FILES: %s does not look like a %s file "
                                         "(LIB_FORMAT is \"%s\")"
                                         % (lib, self.lib_format, self.lib_format))
        self.sim = self.fileset("SIM_FILES", netlist=True)
        self.blackboxes = self.tcl_list("BLACKBOX_MODULES")

        # Tool settings: TOOL_CMD(<tool>), TOOL_SETTINGS(<tool>,<name>), TOOL_HOOKS(<tool>,<point>)
        self.tool_cmd = self.array("TOOL_CMD")
        for tool, cmd in self.tool_cmd.items():
            if tool != FM:
                self.errors.append("TOOL_CMD(%s): unknown tool \"%s\" (supported: %s)"
                                   % (tool, tool, FM))
            elif re.search(r"[\r\n]", cmd):
                self.errors.append("TOOL_CMD(%s): must be one line" % tool)
        self.tool_settings = self.keyed_array("TOOL_SETTINGS")
        self.tool_hooks = {}
        for tool, points in self.keyed_array("TOOL_HOOKS").items():
            for point, file_name in points.items():
                if tool == FM and point not in HOOK_POINTS:
                    self.errors.append("TOOL_HOOKS(%s,%s): unknown hook point (valid: %s)"
                                       % (tool, point, ", ".join(HOOK_POINTS)))
                else:
                    self.tool_hooks.setdefault(tool, {})[point] = self.resolve(
                        "TOOL_HOOKS(%s,%s)" % (tool, point), file_name)
        for name in self.tool_settings.get(FM, {}):
            if not IDENT_RE.match(name):
                self.errors.append("TOOL_SETTINGS(%s,%s): not a valid app var name" % (FM, name))

        # Run environment. An ENV_SOURCE item is a file, optionally
        # followed by arguments: "/tools/setup.cshrc -v 2023.12".
        self.env_source = []
        for item in self.tcl_list("ENV_SOURCE"):
            words = item.split()
            self.env_source.append([self.resolve("ENV_SOURCE", words[0])] + words[1:])
        self.env_setenv = self.array("ENV_SETENV")
        for key in self.env_setenv:
            if not IDENT_RE.match(key):
                self.errors.append("ENV_SETENV(%s): not a valid environment variable name" % key)
        try:
            env_lines(self)          # reports values csh cannot quote
        except ConfigError as exc:
            self.errors.append(str(exc))

    def file_error(self, msg):
        """An error caused by a missing or unreadable input file."""
        self.file_problems += 1
        if self.force:
            self.forced.append(msg)
            self.warnings.append("%s (ignored: --force)" % msg)
        else:
            self.errors.append(msg)

    # Typed getters

    def _get(self, name, kind):
        value = self._raw.get(name)
        if value is None:
            return None
        is_array = isinstance(value, dict)
        if is_array != (kind == "array"):
            self.errors.append("%s must be %s" % (name, "an array, e.g. set %s(key) \"value\"" % name
                                                  if kind == "array"
                                                  else "a plain variable, not an array"))
            return None
        return value

    def scalar(self, name, default=None, required=False):
        value = self._get(name, "scalar")
        if value is None or value.strip() == "":
            if required:
                self.errors.append("%s is required" % name)
            return default
        return value.strip()

    def choice(self, name, options, default):
        value = self.scalar(name, default=default)
        if value not in options:
            self.errors.append("%s: \"%s\" is not one of %s"
                               % (name, value, ", ".join(sorted(options))))
        return value

    def tcl_list(self, name, default=None, required=False):
        """List items, with $variables expanded (braces do not expand them in Tcl)."""
        value = self._get(name, "scalar")
        items = []
        try:
            raw_items = self._tcl.splitlist(value) if value else ()
        except tkinter.TclError as exc:
            self.errors.append("%s: not a valid list (%s). Check the quotes and braces."
                               % (name, exc))
            return list(default or [])
        for item in raw_items:
            try:
                item = str(self._tcl.call("subst", "-nocommands", "-nobackslashes", item))
            except tkinter.TclError as exc:
                self.errors.append("%s: %s" % (name, exc))
                continue
            if item.strip():
                items.append(item.strip())
        if not items:
            if required:
                self.errors.append("%s is required (a list in braces)" % name)
            return list(default or [])
        return items

    def array(self, name):
        return self._get(name, "array") or {}

    def keyed_array(self, name):
        """NAME(<tool>,<key>) -> {tool: {key: value}}."""
        result = {}
        for key, value in self.array(name).items():
            tool, sep, sub = key.partition(",")
            if not sep or not tool or not sub:
                self.errors.append("%s(%s): key must be <tool>,<name>, e.g. %s(%s,...)"
                                   % (name, key, name, FM))
                continue
            if tool.strip() != FM:
                self.errors.append("%s(%s): unknown tool \"%s\" (supported: %s)"
                                   % (name, key, tool.strip(), FM))
                continue
            result.setdefault(tool.strip(), {})[sub.strip()] = value
        return result

    def resolve(self, label, path, base=None):
        """Absolute path (relative to base, default the config dir).

        A missing file is an error, or a warning with --force.
        """
        full = os.path.normpath(os.path.join(base or self.dir, os.path.expanduser(path)))
        if not os.path.exists(full):
            self.file_error("%s: file not found: %s" % (label, full))
        return full

    def paths(self, name, required=False):
        return [self.resolve(name, p) for p in self.tcl_list(name, required=required)]

    def search_dirs(self, name):
        """Directories of a *_SEARCH_DIRS setting (missing ones are file errors)."""
        dirs = []
        for d in self.paths(name):
            if os.path.exists(d) and not os.path.isdir(d):
                self.errors.append("%s: not a directory: %s" % (name, d))
            else:
                add_unique(dirs, [d])
        return dirs

    def find(self, label, path, search, base=None):
        """Locate a file. A full path is used as given. A relative one is
        tried next to base (the .f file, or the config), then in the search
        dirs in order; the first match wins. Not found anywhere is an error,
        or a warning with --force."""
        path = os.path.expanduser(path)
        base = base or self.dir
        if os.path.isabs(path):
            return self.resolve(label, path)
        dirs = []
        add_unique(dirs, [base] + list(search))
        found = []
        add_unique(found, [os.path.normpath(os.path.join(d, path)) for d in dirs
                           if os.path.exists(os.path.join(d, path))])
        if len(found) > 1:
            self.warnings.append("%s: %s found in several places, using %s"
                                 % (label, path, found[0]))
        if found:
            return found[0]
        self.file_error("%s: file not found: %s (searched: %s)" % (label, path, ", ".join(dirs)))
        return os.path.normpath(os.path.join(base, path))

    def expand(self, text):
        """$NAME / ${NAME} in a .f file: config variables first, then
        environment variables. Unknown names are left in place."""
        def value(m):
            name = m.group(1) or m.group(2)
            config_value = self._raw.get(name)
            if isinstance(config_value, str):
                return config_value
            return os.environ.get(name, m.group(0))
        return re.sub(r"\$\{(\w+)\}|\$(\w+)", value, text)

    def fileset(self, name, search_var=None, required=False, netlist=False, incdirs=()):
        """Files from a list variable; entries ending in .f are file lists.

        A file name without a full path is looked up next to the config (or
        next to its .f file), then in the search_var dirs, then in the
        include dirs (incdirs plus +incdir+ from the .f files).
        """
        result = FileSet()
        add_unique(result.incdirs, list(incdirs))
        search = self.search_dirs(search_var) if search_var else []
        problems = self.file_problems
        items = self.tcl_list(name, required=required)
        # Read every .f file first, so +incdir+ dirs from anywhere in them
        # are known before any file name is looked up.
        refs = []                        # (label for messages, file name, base dir)
        for item in items:
            if item.endswith(".f"):
                self.read_filelist(name, self.find(name, item, search), result, refs,
                                   [], search)
            else:
                refs.append((name, item, None))
        if netlist and (result.incdirs or result.defines):
            self.warnings.append("%s: +incdir+/+define+ in a .f file ignored for netlists" % name)
            result.incdirs, result.defines = [], []
        lookup = list(search)
        add_unique(lookup, result.incdirs)
        for label, ref, base in refs:
            add_unique(result.files, [self.find(label, ref, lookup, base)])
        if required and items and not result.files and self.file_problems == problems:
            self.errors.append("%s: the .f file lists contain no files" % name)
        return result

    def read_filelist(self, name, path, result, refs, stack, search):
        """Read a .f file: one name per line, // or # comments.

        A comment starts with // or # at the start of a line or after a
        space. $VAR and ${VAR} are config variables, then environment
        variables. Also understands -f/-F <file> (nested lists), -v <file>,
        +incdir+<dir>[+<dir>...] and +define+<MACRO>[=<value>][+...].
        File names go to refs and are looked up later (see fileset);
        +incdir+ dirs and nested .f files are relative to this .f file.
        """
        label = "%s (%s)" % (name, os.path.basename(path))
        real = os.path.realpath(path)
        if real in stack:
            self.errors.append("%s: file list includes itself" % label)
            return
        if not os.path.exists(path):
            return                       # already reported by find()
        try:
            with open(path, encoding="ascii", errors="replace") as fh:
                lines = fh.readlines()
        except (IOError, OSError) as exc:
            self.file_error("%s: cannot read file list: %s" % (label, exc))
            return
        base = os.path.dirname(path)
        for lineno, line in enumerate(lines, 1):
            line = FILELIST_COMMENT_RE.split(line, 1)[0].strip()
            if not line:
                continue
            where = "%s line %d" % (label, lineno)
            tokens = []
            for tok in line.split():
                tok = self.expand(tok)
                if "$" in tok:
                    self.errors.append("%s: variable not set in the config or the "
                                       "environment: %s" % (where, tok))
                else:
                    tokens.append(tok)
            i = 0
            while i < len(tokens):
                tok = tokens[i]
                if tok in ("-f", "-F", "-v", "-y"):
                    if i + 1 >= len(tokens):
                        self.errors.append("%s: %s needs a file name" % (where, tok))
                        break
                    if tok == "-y":
                        self.warnings.append("%s: -y library directories are not supported, "
                                             "ignored: %s" % (where, tokens[i + 1]))
                    elif tok == "-v":
                        refs.append((where, tokens[i + 1], base))
                    else:
                        self.read_filelist(name, self.find(where, tokens[i + 1], search, base),
                                           result, refs, stack + [real], search)
                    i += 2
                    continue
                if tok.startswith("+incdir+"):
                    add_unique(result.incdirs, [self.resolve(where, d, base)
                                                for d in tok[len("+incdir+"):].split("+") if d])
                elif tok.startswith("+define+"):
                    add_unique(result.defines,
                               [d for d in tok[len("+define+"):].split("+") if d])
                elif tok.startswith(("-", "+")):
                    self.warnings.append("%s: option %s not supported, ignored" % (where, tok))
                else:
                    refs.append((where, tok, base))
                i += 1


# Quoting helpers

FILELIST_COMMENT_RE = re.compile(r"(?:^|\s)(?://|#)")

_TCL_BARE = re.compile(r"^[A-Za-z0-9_./:+=@%,~-]+\Z")
_TCL_SPECIAL = re.compile(r'([\\{}\[\]$";# \t])')


def tcl_word(text):
    """Quote text as one Tcl word (also valid as a Tcl list element)."""
    if _TCL_BARE.match(text):
        return text
    if text and not re.search(r"[{}\\\r\n]", text):
        return "{%s}" % text
    text = _TCL_SPECIAL.sub(r"\\\1", text)
    return text.replace("\n", "\\n").replace("\r", "\\r") or "{}"


def tcl_list_block(items, indent="  "):
    """A braced, one-item-per-line Tcl list."""
    if not items:
        return "{}"
    return "{\n%s\n}" % "\n".join(indent + tcl_word(i) for i in items)


def csh_word(label, text):
    """Single-quote a value for csh; reject what csh cannot quote safely."""
    if re.search(r"['!\r\n]", text):
        raise ConfigError("%s: value may not contain ' or ! or newlines for csh: %r"
                          % (label, text))
    return "'%s'" % text


def csh_env_value(label, text):
    """Double-quote a setenv value for csh, so $VAR is expanded at run time."""
    if re.search(r'["`!\r\n]', text):
        raise ConfigError("%s: value may not contain \" or ` or ! or newlines for csh: %r"
                          % (label, text))
    return '"%s"' % text


def env_lines(cfg):
    """csh lines that set up the environment: ENV_SOURCE, then ENV_SETENV."""
    lines = []
    for words in cfg.env_source:
        path = csh_word("ENV_SOURCE", words[0])
        args = "".join(" " + csh_word("ENV_SOURCE", w) for w in words[1:])
        lines += [
            "if ( ! -r %s ) then" % path,
            "  echo %s" % csh_word("ENV_SOURCE", "ERROR: cannot read " + words[0]),
            "  exit 2",
            "endif",
            "source %s%s" % (path, args),
        ]
    for name, value in cfg.env_setenv.items():
        lines.append("setenv %s %s" % (name, csh_env_value("ENV_SETENV(%s)" % name, value)))
    return lines


# Formality script generation

def section(title):
    return ["", "# " + title]


def hook_lines(cfg, point):
    """'source <file>' for TOOL_HOOKS(formality,<point>), or nothing."""
    hook = cfg.tool_hooks.get(FM, {}).get(point)
    if not hook:
        return []
    return ["", "# Hook: %s" % point, "source %s" % tcl_word(hook)]


def vcs_option(fileset):
    """-vcs option carrying include dirs and defines, or ''."""
    opts = ["+incdir+" + d for d in fileset.incdirs] + ["+define+" + d for d in fileset.defines]
    return " -vcs %s" % tcl_word(" ".join(opts)) if opts else ""


def forced_header(cfg):
    """Comment lines listing the input problems ignored because of --force."""
    if not cfg.forced:
        return []
    return (["# WARNING: generated with --force, these input problems were ignored:"]
            + ["#   " + msg for msg in cfg.forced])


def gen_fm_rtl2syn(cfg, flow):
    """Formality script, <run_dir>/<flow>/<flow>.tcl, run from that directory."""
    settings = dict(cfg.tool_settings.get(FM, {}))
    auto_setup = settings.pop("synopsys_auto_setup", "true")
    lines = ["# Formality %s: RTL vs synthesized netlist for %s" % (flow, cfg.top)]
    lines += forced_header(cfg)
    lines += [
        "",
        "set FLOW %s" % tcl_word(flow),
        "set TOP_DESIGN %s" % tcl_word(cfg.top),
        "file delete -force ${FLOW}.status",
        "file delete -force {*}[glob -nocomplain reports/${FLOW}.*.rpt]",
    ]

    # synopsys_auto_setup first: it changes the defaults of other variables.
    lines += section("Setup")
    lines.append("set_app_var synopsys_auto_setup %s" % tcl_word(auto_setup))
    lines += ["set_app_var %s %s" % (n, tcl_word(settings[n])) for n in sorted(settings)]
    lines += hook_lines(cfg, "setup")

    if cfg.svf:
        lines += section("Synthesis guidance")
        lines.append("set_svf %s" % tcl_word(cfg.svf))

    lines += section("Libraries")
    lines += ["%s %s" % (LIB_READERS[cfg.lib_format], tcl_word(lib)) for lib in cfg.libs]

    lines += section("Reference: RTL")
    lines.append("%s -r %s%s" % (RTL_READERS[cfg.rtl_format], tcl_list_block(cfg.rtl.files),
                                 vcs_option(cfg.rtl)))
    if cfg.sim.files:
        lines.append("read_verilog -r %s" % tcl_list_block(cfg.sim.files))

    lines += section("Implementation: synthesized netlist")
    lines.append("read_verilog -i -netlist %s" % tcl_list_block(cfg.syn.files))
    if cfg.sim.files:
        lines.append("read_verilog -i %s" % tcl_list_block(cfg.sim.files))

    if cfg.blackboxes:
        lines += section("Black boxes")
        for module in cfg.blackboxes:
            lines += ["set_black_box %s" % tcl_word("r:/WORK/" + module),
                      "set_black_box %s" % tcl_word("i:/WORK/" + module)]

    lines += section("Top-level designs")
    lines += ["set_top r:/WORK/$TOP_DESIGN",
              "set_top i:/WORK/$TOP_DESIGN"]
    lines += hook_lines(cfg, "pre_match")

    lines += section("Match")
    lines += ["match",
              "report_unmatched_points > reports/${FLOW}.unmatched.rpt"]
    lines += hook_lines(cfg, "pre_verify")

    # The status file is written right after verify, so a problem in a
    # report or hook below cannot turn a finished run into "did not finish".
    lines += section("Verify")
    lines += [
        "set verify_ok [verify]",
        "if {$verify_ok} {",
        "  set result PASS",
        "} else {",
        "  set result FAIL",
        "}",
        "set fh [open ${FLOW}.status w]",
        "puts $fh $result",
        "close $fh",
        "if {!$verify_ok} {",
        "  report_failing_points > reports/${FLOW}.failing.rpt",
        "  report_aborted_points > reports/${FLOW}.aborted.rpt",
        "  analyze_points -failing > reports/${FLOW}.analyze.rpt",
        "}",
    ]
    lines += hook_lines(cfg, "post_verify")

    lines += section("Finish")
    lines += [
        "save_session -replace ${FLOW}",
        'puts "$FLOW result: $result"',
        "exit",
        "",
    ]
    return "\n".join(lines)


FLOW_GENERATORS = {"rtl2syn": gen_fm_rtl2syn}


def gen_run_script(cfg, flow):
    tool_cmd = cfg.tool_cmd.get(FM, "").strip() or "fm_shell"
    lines = [
        "#!/bin/tcsh -f",
        "# Run Formality %s for %s" % (flow, cfg.top),
        "# Exit status: 0 PASS, 1 FAIL, 2 did not finish (setup or tool problem)",
    ]
    lines += forced_header(cfg)
    lines += [
        "",
        "set flow = %s" % flow,
        'set here = "`dirname $0`"',
        'if ( ! -d "$here/$flow" ) then',
        '  echo "ERROR: directory not found: $here/$flow"',
        "  exit 2",
        "endif",
        'cd "$here/$flow"',
        "",
        "# Environment",
    ]
    lines += env_lines(cfg)
    lines += [
        "",
        "# Run the tool. Input from /dev/null so it exits if the script stops early.",
        "# The old result is removed first, so a tool that fails to start is not",
        "# reported with the result of an earlier run.",
        "rm -f ${flow}.status",
        "%s -f ${flow}.tcl < /dev/null |& tee logs/${flow}.log" % tool_cmd,
        "",
        "if ( ! -e ${flow}.status ) then",
        '  echo "$flow did not finish. See $flow/logs/${flow}.log"',
        "  exit 2",
        "endif",
        'set result = "`cat ${flow}.status`"',
        'echo "$flow result: $result (logs: $flow/logs, reports: $flow/reports)"',
        'if ( "$result" != "PASS" ) exit 1',
        "exit 0",
        "",
    ]
    return "\n".join(lines)


# Main

def write_file(path, text, executable=False):
    with open(path, "w", encoding="ascii", newline="\n") as fh:
        fh.write(text)
    if executable:
        mode = os.stat(path).st_mode
        os.chmod(path, mode | stat.S_IXUSR | stat.S_IXGRP | stat.S_IXOTH)


def block_names():
    """Blocks that have a config in configs/ (templates not counted)."""
    if not os.path.isdir(CONFIG_DIR):
        return []
    return sorted(n[:-len(CONFIG_SUFFIX)] for n in os.listdir(CONFIG_DIR)
                  if n.endswith(CONFIG_SUFFIX) and n != PROJECT_CONFIG
                  and n not in TEMPLATE_CONFIGS)


def find_config(name):
    """Config file for a block name (or a path to a config file)."""
    if os.path.isfile(name):
        return name
    path = os.path.join(CONFIG_DIR, name + CONFIG_SUFFIX)
    if os.path.isfile(path):
        return path
    blocks = block_names()
    raise ConfigError("no config for \"%s\": %s not found.\n"
                      "flowgen: blocks with a config: %s\n"
                      "flowgen: to start a new block: %s --new %s"
                      % (name, path, ", ".join(blocks) or "none yet", WRAPPER, name))


def new_block(block):
    """Create configs/<block>.config.tcl (and the project config if missing)."""
    if not IDENT_RE.match(block):

#!/usr/bin/env python3
"""pack - concatenate a repo's files into one plain-text bundle.

Usage:
    pack.py [-o BUNDLE] [ROOT]

ROOT defaults to the git repo containing the current directory (or the
current directory if not in git). In a git repo only tracked files are
packed, as they are on disk, with the file mode recorded in git; files
that are not committed are listed in a warning. Outside a git repo every
file under ROOT is packed. Paths under EXCLUDE_DIRS are never packed:
docs/, and common_files/ (local files that live only on the user's
machine).

The bundle is pure ASCII, so it survives copy/paste, downloads and CRLF
line endings. Pure-ASCII text files are stored as readable lines; any other
file (non-ASCII text, binary) is stored as base64 and restored byte-exact.
Recreate the files with unpack.py (which is itself inside the bundle).

Bundle format:
    #@@PACK 3
    #@@FILE <path> mode=<octal> enc=<text|b64> lines=<n> eol=<0|1> md5=<hex>
    <n lines: file content (text) or base64 of the raw bytes (b64)>
    #@@END <path>

Python 3.6+, standard library only.
"""

import argparse
import base64
import hashlib
import os
import re
import subprocess
import sys

EXCLUDE_DIRS = ("docs/", "common_files/")   # top-level directories never packed
PATH_RE = re.compile(r"^[A-Za-z0-9._+-]+(/[A-Za-z0-9._+-]+)*\Z")   # what unpack.py accepts


class PackError(Exception):
    pass


def git(root, *args):
    return subprocess.check_output(("git", "-C", root) + args,
                                   stderr=subprocess.DEVNULL).decode("utf-8", "replace")


def repo_root():
    try:
        out = subprocess.check_output(["git", "rev-parse", "--show-toplevel"],
                                      stderr=subprocess.DEVNULL)
        return out.decode().strip()
    except (OSError, subprocess.CalledProcessError):
        return os.getcwd()


def in_git_repo(root):
    path = root
    while True:
        if os.path.exists(os.path.join(path, ".git")):
            return True
        parent = os.path.dirname(path)
        if parent == path:
            return False
        path = parent


def list_files(root):
    """Return ({relative path: mode}, [warnings]); paths use '/' separators.

    In a git repo: the tracked files, with the mode recorded in git (so a
    bundle packed on Windows keeps the execute bits).
    """
    modes, warnings = {}, []
    if in_git_repo(root):
        try:
            # Each entry: "<mode> <hash> <stage>\t<path>"
            for entry in git(root, "ls-files", "-s", "-z").split("\0"):
                if entry:
                    info, _, rel = entry.partition("\t")
                    modes[rel] = 0o755 if info.startswith("100755") else 0o644
            # Each entry: "XY <path>"
            for entry in git(root, "status", "--porcelain", "-z").split("\0"):
                rel = entry[3:]
                if not entry or rel.startswith(EXCLUDE_DIRS):
                    continue
                if entry.startswith("??"):
                    warnings.append("not in git, NOT packed: %s" % rel)
                elif os.path.isfile(os.path.join(root, rel)):
                    warnings.append("not committed, packed as it is on disk: %s" % rel)
                else:
                    warnings.append("deleted on disk, NOT packed: %s" % rel)
        except (OSError, subprocess.CalledProcessError) as exc:
            raise PackError("%s is a git repo but git failed (%s); nothing packed" % (root, exc))
    else:
        for dirpath, dirnames, filenames in os.walk(root):
            dirnames[:] = [d for d in dirnames if d != ".git"]
            for name in filenames:
                path = os.path.join(dirpath, name)
                rel = os.path.relpath(path, root).replace(os.sep, "/")
                modes[rel] = 0o755 if os.stat(path).st_mode & 0o100 else 0o644
    modes = dict((f, m) for f, m in modes.items()
                 if os.path.isfile(os.path.join(root, f)) and not f.startswith(EXCLUDE_DIRS))
    return modes, warnings


def main(argv=None):
    parser = argparse.ArgumentParser(description="Pack repo files into one text bundle.")
    parser.add_argument("root", nargs="?", help="directory to pack (default: git repo root)")
    parser.add_argument("-o", "--output", help="bundle file (default: bundle.txt)")
    args = parser.parse_args(argv)

    root = os.path.abspath(args.root or repo_root())
    output = os.path.abspath(args.output or "bundle.txt")
    if not os.path.isdir(root):
        sys.exit("pack: ERROR: not a directory: %s" % root)
    try:
        modes, warnings = list_files(root)
    except PackError as exc:
        sys.exit("pack: ERROR: %s" % exc)

    chunks = ["#@@PACK 3\n"]
    packed, encoded = 0, []
    for rel in sorted(modes):
        path = os.path.join(root, rel)
        if os.path.abspath(path) == output:
            continue
        if not PATH_RE.match(rel):
            sys.exit("pack: ERROR: file name not supported (use letters, digits and "
                     "._+- only): %s" % rel)
        with open(path, "rb") as fh:
            data = fh.read()
        if data.startswith(b"#@@PACK"):
            continue                     # never pack a bundle inside a bundle
        mode = modes[rel]

        norm = data.replace(b"\r\n", b"\n")
        if all(b in b"\t\n" or 32 <= b < 127 for b in norm):
            enc, data = "text", norm
            text = data.decode("ascii")
            eol = 1 if text.endswith("\n") or not text else 0
            lines = text.split("\n")
            if eol:
                lines = lines[:-1]       # drop the empty piece after the last newline
        else:
            enc, eol = "b64", 1
            b64 = base64.b64encode(data).decode("ascii")
            lines = [b64[i:i + 76] for i in range(0, len(b64), 76)]
            encoded.append(rel)
        md5 = hashlib.md5(data).hexdigest()

        chunks.append("#@@FILE %s mode=%03o enc=%s lines=%d eol=%d md5=%s\n"
                      % (rel, mode, enc, len(lines), eol, md5))
        chunks.extend(line + "\n" for line in lines)
        chunks.append("#@@END %s\n" % rel)
        packed += 1

    if not packed:
        sys.exit("pack: ERROR: no files to pack in %s; %s not written" % (root, output))
    tmp = output + ".tmp"
    with open(tmp, "w", encoding="ascii", newline="\n") as fh:
        fh.writelines(chunks)
    os.replace(tmp, output)             # an old bundle is only replaced by a complete one

    for msg in warnings:
        print("pack: WARNING: %s" % msg)
    for rel in encoded:
        print("pack: note: not pure ASCII text, stored as base64: %s" % rel)
    print("pack: %d files from %s -> %s" % (packed, root, output))
    return 0


if __name__ == "__main__":
    sys.exit(main())
#@@END pack.py

  uncommitted files are listed as warnings. Errors (bundle not touched):

  its lines, then `#@@END <path>`. Pure-ASCII text files are stored as
  readable lines; anything else as base64. This survives charset
  conversion and CRLF line endings (a non-ASCII bundle was once corrupted

## flowgen.py

Usage:
```
runflowgen.sh --new BLOCK
runflowgen.sh [--force] BLOCK
```
- --new BLOCK: writes configs/BLOCK.config.tcl from
  configs/new_block.config.tcl and, if missing,
  configs/project.config.tcl from configs/project_template.config.tcl,
  then prints what to edit and the generate command. Never overwrites.
- BLOCK: reads configs/BLOCK.config.tcl (configs/ is found next to
  the running flowgen.py). A path to
  a config file is also accepted. With an unknown block the error lists
  the blocks that have a config.
- After generating, flowgen prints the run script to launch and where
  the result and log will be. The project config is copied into the run
  directory next to the block config.
- Missing or unreadable input files (RTL, netlist, library, SVF, hook,
  cshrc, .f list) are errors. With --force they become warnings and the
  run directory is generated anyway; the ignored problems are listed in
  the headers of the generated scripts. Other config errors (missing
  settings, bad values, Tcl errors) still stop it. Files of settings no
  flow reads yet (APR_FILES, SDC_FILES, UPF_FILES) are not checked.

Output (in the current directory; never overwritten, a same-second rerun
gets a _2, _3 suffix):
```
<yyyy_mm_dd_hh_mm_ss>_<TOP_DESIGN>_formality/
  block.config.tcl      copy of the config used (original file name)
  run_rtl2syn.csh       tcsh: source ENV_SOURCE, setenv ENV_SETENV, run tool
  rtl2syn/              the tool runs here
    rtl2syn.tcl         generated Formality script
    logs/               empty; log goes to logs/rtl2syn.log
    reports/            empty; reports go here
```
- run_rtl2syn.csh runs `<TOOL_CMD> -f rtl2syn.tcl < /dev/null |& tee
  logs/rtl2syn.log` from rtl2syn/, then reads rtl2syn/rtl2syn.status.
  Exit codes: 0 PASS, 1 FAIL, 2 did not finish. The run script removes
  the old status file before starting the tool and exits 2 if an
  ENV_SOURCE file cannot be read.
- Generated scripts are kept simple: one-line header, no generation date
  or "generated by" comments, no placeholder comments, 2-space indent.

How it works:
- The config is sourced in a real Tcl interpreter (tkinter.Tcl()) with
  the working directory set to the config's real directory. Only global
  variables created by the config are read; arrays become dicts.
- Config path is resolved with realpath, so relative paths are relative
  to the config's real directory (flowgen/configs/).
- `$VARIABLES` inside braced list items are expanded by flowgen
  (Tcl `subst -nocommands -nobackslashes`).
- All values are validated; all errors are reported together; nothing is
  written unless everything renders. Unknown names close to a known
  setting (difflib, e.g. BLACK_BOX_MODULES, top_design) produce a typo
  warning. A tool key other than "formality" is an error.
- .f file lists are expanded by flowgen: one path per line, // and #
  comments (line start or after a space), $VAR from the config variables,
  then the environment (unset is an error), relative to the .f file. Supports nested -f/-F,
  -v, +incdir+, +define+. Duplicates are dropped. -y and other options
  are ignored with a warning.
- File name lookup (RTL_FILES, SYN_FILES, names inside .f files): next to
  the config or the .f file first, then *_SEARCH_DIRS in order, then (RTL
  only) RTL_INCDIRS and +incdir+ dirs. All .f files are read before any
  lookup. First match wins (warning if several); not found anywhere is a
  file error listing the dirs searched.
- ENV_SETENV entries are written in config order with double quotes
  ($VAR expanded by csh at run time). ENV_SOURCE items may carry
  arguments after the file name.

#!/usr/bin/env python3
"""unpack - recreate files from a bundle made by pack.py.

Usage:
    unpack.py BUNDLE [-d DIR] [--force] [--list]

Files are written under DIR (default: current directory). Your block
configs live in DIR/flowgen/configs/, next to the templates; flowgen reads
them from there, so unpack never has to be run again for a new config.

If a top-level directory from the bundle (flowgen/) already exists in DIR,
unpack stops with an error and changes nothing. With --force that directory
is removed first and recreated from the bundle, so no stale files are left
behind. If it holds files of your own (anything that is not in the bundle,
e.g. your configs or run directories), it is not deleted but renamed to
flowgen.old_<date>_<time>; copy your configs back from its configs/. Nothing else in DIR is removed:
common_files/ is never in a bundle and is never touched. Top-level files
(pack.py, unpack.py) are overwritten.

Every file is checked against its MD5 before anything is written; a damaged
bundle writes nothing.

Python 3.6+, standard library only.
"""

import argparse
import base64
import binascii
import datetime
import hashlib
import os
import re
import shutil
import sys

CONFIG_SRC = "common_files"          # local golden configs (never in a bundle)

PROTECTED_TOPS = (CONFIG_SRC, ".git")  # a bundle may never write here
PATH_RE = re.compile(r"^[A-Za-z0-9._+-]+(/[A-Za-z0-9._+-]+)*\Z")

HEADER_RE = re.compile(r"^#@@FILE (\S+) mode=([0-7]{3}) enc=(text|b64) lines=(\d+) "
                       r"eol=([01]) md5=([0-9a-f]{32})$")


class BundleError(Exception):
    pass


def read_bundle(bundle_path):
    with open(bundle_path, "rb") as fh:
        raw = fh.read()
    if raw.startswith(b"\xef\xbb\xbf"):
        raw = raw[3:]                   # BOM added by some editors
    try:
        return raw.decode("ascii")
    except UnicodeDecodeError as exc:
        raise BundleError("bundle has a non-ASCII byte 0x%02x at offset %d: it was changed "
                          "during transfer (character-set conversion). Bundles are pure "
                          "ASCII; copy it again as-is (e.g. download the raw file)."
                          % (raw[exc.start], exc.start))


def parse(bundle_path):
    """Return [(path, mode, bytes)] after validating the whole bundle."""
    lines = read_bundle(bundle_path).splitlines()
    if not lines or lines[0] != "#@@PACK 3":
        raise BundleError("not a pack.py bundle, or made by a different pack.py version "
                          "(first line must be '#@@PACK 3')")

    files, seen, i = [], set(), 1
    while i < len(lines):
        if lines[i].strip() == "":
            i += 1                      # tolerate blank lines between files
            continue
        m = HEADER_RE.match(lines[i].rstrip())
        if not m:
            raise BundleError("line %d: expected #@@FILE header, got: %r" % (i + 1, lines[i][:80]))
        path, mode, enc, count, eol, md5 = m.groups()
        count = int(count)
        check_path(path, i)
        if path in seen:
            raise BundleError("line %d: file is in the bundle twice: %s" % (i + 1, path))
        seen.add(path)

        body = lines[i + 1:i + 1 + count]
        end = i + 1 + count
        if len(body) != count or end >= len(lines) or lines[end].rstrip() != "#@@END " + path:
            raise BundleError("%s: truncated or damaged (missing #@@END)" % path)

        if enc == "text":
            data = ("\n".join(body) + ("\n" if eol == "1" and count else "")).encode("ascii")
        else:
            try:
                data = base64.b64decode("".join(l.strip() for l in body), validate=True)
            except (binascii.Error, ValueError):
                raise BundleError("%s: damaged base64 data" % path)
        if hashlib.md5(data).hexdigest() != md5:
            raise BundleError("%s: MD5 mismatch (content was changed in transit)" % path)
        files.append((path, int(mode, 8), data))
        i = end + 1
    if not files:
        raise BundleError("the bundle contains no files")
    for path in seen:
        parent = path.rpartition("/")[0]
        while parent:
            if parent in seen:
                raise BundleError("%s is both a file and a directory in the bundle" % parent)
            parent = parent.rpartition("/")[0]
    return files


def check_path(path, i):
    parts = path.split("/")
    if (not PATH_RE.match(path) or "." in parts or ".." in parts
            or parts[0] in PROTECTED_TOPS):
        raise BundleError("line %d: unsafe path: %s" % (i + 1, path))


def own_files(top_dir, bundle_paths, top):
    """Files under top_dir that are the user's: not in the bundle, not links, not caches.

    Links are skipped because older versions of unpack created config
    links in flowgen/configs/.
    """
    found = []
    for dirpath, dirnames, filenames in os.walk(top_dir):
        dirnames[:] = [d for d in dirnames if d != "__pycache__"]
        for name in filenames:
            full = os.path.join(dirpath, name)
            rel = top + "/" + os.path.relpath(full, top_dir).replace(os.sep, "/")
            if not os.path.islink(full) and rel not in bundle_paths and not name.endswith(".pyc"):
                found.append(rel)
    return sorted(found)


def main(argv=None):
    parser = argparse.ArgumentParser(description="Recreate files from a pack.py bundle.")
    parser.add_argument("bundle", help="bundle file made by pack.py")
    parser.add_argument("-d", "--dir", default=".", help="output directory (default: .)")
    parser.add_argument("--force", action="store_true",
                        help="replace the existing flowgen/ (top-level directories in the "
                             "bundle); it is kept as flowgen.old_<date>_<time> if it holds "
                             "files of your own. Nothing else is removed")
    parser.add_argument("--list", action="store_true", help="only list the files")
    args = parser.parse_args(argv)

    try:
        files = parse(args.bundle)
    except (OSError, BundleError) as exc:
        sys.exit("unpack: ERROR: %s" % exc)

    if args.list:
        for path, mode, _ in files:
            print("%03o %s" % (mode, path))
        return 0

    # Top-level directories (flowgen) this bundle owns.
    tops = sorted(set(p.split("/")[0] for p, _, _ in files if "/" in p))
    existing = [t for t in tops if os.path.lexists(os.path.join(args.dir, t))]
    if existing and not args.force:
        sys.exit("unpack: ERROR: already exists in %s: %s\n"
                 "unpack: use --force to remove and recreate them."
                 % (os.path.abspath(args.dir), ", ".join(existing)))
    for path, _, _ in files:
        if "/" not in path and os.path.isdir(os.path.join(args.dir, path)):
            sys.exit("unpack: ERROR: %s is a directory in %s but a file in the bundle"
                     % (path, os.path.abspath(args.dir)))

    bundle_paths = set(p for p, _, _ in files)
    try:
        for top in existing:
            target = os.path.join(args.dir, top)
            if os.path.isdir(target) and not os.path.islink(target):
                mine = own_files(target, bundle_paths, top)
                if mine:
                    backup = "%s.old_%s" % (target, datetime.datetime.now().strftime(
                        "%Y%m%d_%H%M%S"))
                    os.rename(target, backup)
                    print("unpack: existing %s holds %d file(s) of your own (%s%s); kept as %s"
                          % (target, len(mine), ", ".join(mine[:3]),
                             ", ..." if len(mine) > 3 else "", backup))
                else:
                    print("unpack: removing existing %s" % target)
                    shutil.rmtree(target)
            else:
                print("unpack: removing existing %s" % target)
                os.remove(target)

        for path, mode, data in files:
            dest = os.path.join(args.dir, *path.split("/"))
            if os.path.dirname(dest):
                os.makedirs(os.path.dirname(dest), exist_ok=True)
            with open(dest, "wb") as fh:
                fh.write(data)
            os.chmod(dest, mode)
    except OSError as exc:
        sys.exit("unpack: ERROR: %s\nunpack: stopped part way; fix the problem and "
                 "run unpack again with --force." % exc)

    out_dir = os.path.abspath(args.dir)
    print("unpack: %d files -> %s" % (len(files), out_dir))
    wrapper = os.path.join(out_dir, "flowgen", "runflowgen.sh")
    if os.path.isfile(wrapper):
        print("unpack: next, start a block:  %s --new <block>" % wrapper)
        print("unpack:   then generate it:   %s <block>" % wrapper)
    return 0


if __name__ == "__main__":
    sys.exit(main())
