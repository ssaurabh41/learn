# common_files

Your golden block configs, named `<block_name>.config.tcl`, plus the tools
to make them. On your machine this directory is the source of truth:
`unpack.py` only writes the shipped files below and never removes or
changes anything else here. Every `*.config.tcl` here is linked into
`flowgen/configs/`.

Shipped files:

- `block_template.config.csv`  spreadsheet template; every setting, documented
- `block_template.config.tcl`  the same template as a config (made from the CSV)
- `csv2config.py`              turns `<block>.config.csv` into `<block>.config.tcl`

New block:

```
cp block_template.config.csv myblock.config.csv
```

Edit `myblock.config.csv` in LibreOffice (when opening, set the Value
column to Text; save as CSV), then:

```
./csv2config.py myblock.config.csv
```

This writes `myblock.config.tcl`. It will not overwrite an existing file
unless you add `--force`.

#@@FILE common_files/csv2config.py mode=755 enc=text lines=194 eol=1 md5=901bb4624eb2036373aaf266b8ade7f1
#!/usr/bin/env python3
"""csv2config - turn a block config CSV into a block config Tcl file.

Usage:
    csv2config.py [--force] [-o OUTPUT] CSV

Arguments:
    CSV       The block config spreadsheet, named <block_name>.config.csv.
              Start from block_template.config.csv and edit it in
              LibreOffice (keep it as CSV when saving).
    -o        Output file. Default: the CSV name with .csv replaced by
              .tcl, next to the CSV (block.config.csv -> block.config.tcl).
    --force   Overwrite the output file if it already exists.
    -h        Show this help and exit.

Example:
    csv2config.py block.config.csv
    flowgen.py configs/block.config.tcl

CSV columns: Setting, Value, Required, Description, Example.
  - One value per row. For a list, add more rows with the same Setting
    (or an empty Setting cell); they are collected in order.
  - A Setting starting with # is a heading.
  - Empty Value means not used. Empty TOOL_*/ENV_SETENV entries are left out.
  - Description becomes a comment above the setting. Required and Example
    are for the reader only.
  - Settings flowgen does not know are written as your own helper
    variables, so you can use them as $NAME in other values.

Python 3.6+, standard library only.
"""

import argparse
import csv
import os
import re
import sys
import textwrap

# Settings that are lists in the config (one CSV row per item).
LIST_SETTINGS = (
    "FLOWS", "RTL_SEARCH_DIRS", "RTL_FILES", "RTL_INCDIRS", "RTL_DEFINES",
    "SYN_SEARCH_DIRS", "SYN_FILES", "APR_FILES", "SDC_FILES", "UPF_FILES",
    "LIB_FILES", "SIM_FILES", "BLACKBOX_MODULES", "ENV_SOURCE",
)
NAME_RE = re.compile(r"^[A-Za-z_][A-Za-z0-9_]*(\([^()\s]+\))?$")
COLUMNS = ("setting", "value", "required", "description", "example")


class CsvError(Exception):
    pass


def tcl_quote(text):
    """Text in double quotes for Tcl. $NAME stays live (variables expand);
    characters Tcl would treat specially otherwise are escaped."""
    for ch in ("\\", '"', "[", "]"):
        text = text.replace(ch, "\\" + ch)
    return '"%s"' % text


def read_rows(path):
    """Rows as dicts with the COLUMNS keys, plus the CSV line number."""
    try:
        with open(path, "rb") as fh:
            raw = fh.read()
    except (IOError, OSError) as exc:
        raise CsvError("cannot read %s: %s" % (path, exc))
    try:
        text = raw.decode("utf-8-sig")
    except UnicodeDecodeError:
        raise CsvError("%s is not UTF-8/ASCII text; save it as CSV with UTF-8" % path)
    lines = text.splitlines(True)
    if not lines:
        raise CsvError("%s is empty" % path)
    # LibreOffice may save with ; or tab instead of , depending on settings.
    header = lines[0]
    delimiter = max((",", ";", "\t"), key=header.count)
    reader = csv.reader(lines, delimiter=delimiter)
    head = [h.strip().lower() for h in next(reader)]
    if head[:2] != ["setting", "value"]:
        raise CsvError("%s: first row must be the header Setting, Value, ... (got %s)"
                       % (path, ", ".join(head)))
    rows = []
    for lineno, cells in enumerate(reader, 2):
        cells = [c.strip() for c in cells] + [""] * len(COLUMNS)
        row = dict(zip(COLUMNS, cells))
        row["line"] = lineno
        rows.append(row)
    return rows


def collect(rows):
    """Group rows into ordered entries: ('heading', title) or
    ('setting', name, [values], description)."""
    entries, by_name, current = [], {}, None
    errors = []
    for row in rows:
        name, value = row["setting"], row["value"]
        if not any(row[c] for c in COLUMNS):
            continue
        if name.startswith("#"):
            entries.append(("heading", name.lstrip("#").strip()))
            current = None
            continue
        if not name:
            if not value:
                continue
            if current is None:
                errors.append("line %d: value %r has no Setting above it" % (row["line"], value))
                continue
            name = current
        if not NAME_RE.match(name):
            errors.append("line %d: %r is not a valid setting name" % (row["line"], name))
            continue
        if name in by_name:
            entry = by_name[name]
            if name not in LIST_SETTINGS:
                errors.append("line %d: %s is a single value but appears again "
                              "(first on line %d)" % (row["line"], name, entry[4]))
                continue
        else:
            entry = ["setting", name, [], row["description"], row["line"]]
            by_name[name] = entry
            entries.append(entry)
        if value:
            entry[2].append(value)
        current = name
    if errors:
        raise CsvError("\n".join(errors))
    return entries


def render(entries, csv_name):
    blocks, heading = [], None        # heading is written only if a setting follows
    for entry in entries:
        if entry[0] == "heading":
            heading = entry[1]
            continue
        _, name, values, description, _ = entry
        if "(" in name and not values:
            continue                      # empty TOOL_*/ENV_SETENV rows are left out
        lines = []
        if heading is not None:
            lines += ["", "# " + heading]
            heading = None
        lines.append("")
        for para in description.splitlines():
            lines += ["# " + l for l in textwrap.wrap(para, 74)]
        if name in LIST_SETTINGS:
            if values:
                lines.append("set %s {" % name)
                lines += ["  " + tcl_quote(v) for v in values]
                lines.append("}")
            else:
                lines.append("set %s {}" % name)
        else:
            lines.append("set %s %s" % (name, tcl_quote(values[0] if values else "")))
        blocks += lines
    header = "# From %s. Edit the CSV and rerun csv2config.py." % csv_name
    return "\n".join([header] + blocks) + "\n"


def main(argv=None):
    parser = argparse.ArgumentParser(
        description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter)
    parser.add_argument("csv", metavar="CSV", help="block config CSV, <block_name>.config.csv")
    parser.add_argument("-o", "--output", help="output .config.tcl file")
    parser.add_argument("--force", action="store_true", help="overwrite an existing output file")
    args = parser.parse_args(argv)

    if not args.csv.endswith(".config.csv"):
        print("csv2config: WARNING: config CSVs are named <block_name>.config.csv: %s" % args.csv)
    output = args.output or (args.csv[:-len(".csv")] + ".tcl" if args.csv.endswith(".csv")
                             else args.csv + ".tcl")
    if os.path.exists(output) and not args.force:
        sys.exit("csv2config: ERROR: %s already exists. Use --force to overwrite it." % output)

    try:
        entries = collect(read_rows(args.csv))
    except CsvError as exc:
        sys.exit("\n".join("csv2config: ERROR: " + l for l in str(exc).splitlines()))
    text = render(entries, os.path.basename(args.csv))
    if any(ord(c) > 127 for c in text):
        print("csv2config: WARNING: non-ASCII characters in the CSV were copied to %s" % output)

    with open(output, "w") as fh:
        fh.write(text)
    print("csv2config: wrote %s" % output)
    return 0


if __name__ == "__main__":
    sys.exit(main())

#!/bin/tcsh -f
# Regression test for flowgen, using the template config and the example
# design in this directory. Usage: examples/run_regression.csh
#
# Checks:
#   1. csv2config turns block_template.config.csv into exactly the shipped
#      block_template.config.tcl
#   2. flowgen runs on the template config with no warnings or errors
#   3. the run directory has every expected file, and the run script parses
# Output goes to examples/regression_out/ (removed at the start of each run).

set here = "`dirname $0`"
set root = "`cd $here/.. && pwd`"
if ( "$root" == "" || ! -d "$root/flowgen" ) then
  echo "regression: cannot find the repo root from $0"
  exit 2
endif
set out = "$root/examples/regression_out"
if ( -d "$out" ) rm -rf "$out"
mkdir -p "$out"
set fail = 0

# 1. CSV template -> config template
python3 "$root/common_files/csv2config.py" "$root/common_files/block_template.config.csv" \
  -o "$out/block_template.config.tcl" > "$out/csv2config.log"
if ( $status != 0 ) then
  echo "FAIL csv2config (see $out/csv2config.log)"
  set fail = 1
else
  cmp -s "$out/block_template.config.tcl" "$root/common_files/block_template.config.tcl"
  if ( $status != 0 ) then
    echo "FAIL block_template.config.tcl is out of date; regenerate it from the CSV:"
    echo "     common_files/csv2config.py --force common_files/block_template.config.csv"
    set fail = 1
  else
    echo "PASS csv2config"
  endif
endif

# 2. flowgen on the template config
cd "$out"
python3 "$root/flowgen/flowgen.py" "$root/common_files/block_template.config.tcl" >& flowgen.log
if ( $status != 0 ) then
  echo "FAIL flowgen exited with an error (see $out/flowgen.log)"
  set fail = 1
else
  grep -q -e "WARNING" -e "ERROR" flowgen.log
  if ( $status == 0 ) then
    echo "FAIL flowgen printed warnings (see $out/flowgen.log)"
    set fail = 1
  else
    echo "PASS flowgen"
  endif
endif

# 3. Run directory contents
set rundir = "`ls -d $out/*_block_formality |& head -1`"
set dirfail = 0
foreach f ( run_rtl2syn.csh rtl2syn/rtl2syn.tcl rtl2syn/logs rtl2syn/reports block_template.config.tcl )
  if ( ! -e "$rundir/$f" ) then
    echo "FAIL missing $f in the run directory"
    set dirfail = 1
  endif
end
if ( -e "$rundir/run_rtl2syn.csh" ) then
  tcsh -n "$rundir/run_rtl2syn.csh"
  if ( $status != 0 ) then
    echo "FAIL run_rtl2syn.csh does not parse"
    set dirfail = 1
  endif
endif
if ( $dirfail == 0 ) then
  echo "PASS run directory"
else
  set fail = 1
endif

if ( $fail != 0 ) then
  echo "regression: FAILED"
  exit 1
endif
echo "regression: PASSED"
exit 0
#@@END examples/run_regression.csh

# flowgen

Generates EDA run directories from one Tcl block config. Python 3.6+ (with tkinter), standard library only.

The first supported tool is Synopsys Formality, with the `rtl2syn` flow: RTL reference vs. synthesized netlist.

## Usage

```
runflowgen.sh configs/block.config.tcl
```

`runflowgen.sh` is a csh (tcsh) wrapper that runs `flowgen.py` with `python3`, passing all arguments through. Add `--force` to generate even when input files are missing or a required file list is empty.

This creates the following in the current directory. If the directory already exists, it is overwritten.

```
<yyyy_mm_dd_hh_mm_ss>_<TOP_DESIGN>_formality/
  block.config.tcl      copy of the config used
  run_rtl2syn.csh       run script (tcsh): sets up the env, runs fm_shell
  rtl2syn/              where Formality runs
    rtl2syn.tcl         generated Formality script
    logs/               empty, filled in by the tool
    reports/            empty, filled in by the tool
```

Nothing is launched. To run a flow:

```
cd <yyyy_mm_dd_hh_mm_ss>_<TOP_DESIGN>_formality
./run_rtl2syn.csh
```

The log goes to `rtl2syn/logs/rtl2syn.log`, the reports to `rtl2syn/reports/`, and the result (PASS or FAIL) to `rtl2syn/rtl2syn.status`. The run script exits with 0 for PASS, 1 for FAIL, and 2 if the flow did not finish.

## Config

Configs are named `<block_name>.config.tcl` and live in `common_files/`. The easy way to make one is the spreadsheet: copy `common_files/block_template.config.csv` to `<block>.config.csv`, edit it in LibreOffice, and run `common_files/csv2config.py <block>.config.csv`. You can also edit the `.tcl` directly; it is plain Tcl.

- Single values go in double quotes: `set TOP_DESIGN "block"`.
- Lists go in braces, one double-quoted item per line. flowgen expands `$VARIABLES` inside list items.
- Relative paths are resolved from the config file's real directory (`common_files/`, even when run through a `flowgen/configs/` link).
- File lists (`RTL_FILES`, `SYN_FILES`, `APR_FILES`, `SIM_FILES`) accept `.f` files. A `.f` file holds one path per line, with `//` or `#` comments. It may also contain `-f`, `-v`, `+incdir+` and `+define+`.
- File names without a full path in `RTL_FILES`, `SYN_FILES` and their `.f` lists are looked up in `RTL_SEARCH_DIRS` / `SYN_SEARCH_DIRS`, in order, then (for RTL) in `RTL_INCDIRS` and `+incdir+` dirs, then next to the config (or the `.f` file).
- `$NAME` inside a `.f` file uses config variables (e.g. your own `USER_RTL_DIR`), then environment variables.
- Helper variables such as `PROJ_ROOT` are optional; flowgen never needs them. A name found in several directories uses the first one, with a warning. A file listed twice is read once.
- Misspelled setting names and missing files produce warnings. Errors are all reported together, and nothing is generated if there are any.

`unpack.py` creates `flowgen/configs/` with a link to each `*.config.tcl` in `common_files/` (the template included); re-run it with `--force` after adding a config. In `common_files/`, `unpack.py` only writes the shipped templates and `csv2config.py`; your own configs are never removed or changed.

## Regression

`examples/` holds a small placeholder design that the template config points to. `examples/run_regression.csh` checks that the CSV template converts to the shipped `.tcl` template, that flowgen runs on it with no warnings, and that the run directory is complete.

class ConfigError(Exception):
    pass


# Reading the Tcl config

def load_tcl_config(path):
    """Source the config in a fresh Tcl interpreter.

    Returns ({name: str or {key: str}}, interpreter) for the global
    variables the config created (arrays become dicts).
    """
    tcl = tkinter.Tcl()
    before = set(tcl.splitlist(tcl.eval("info globals")))

    old_cwd = os.getcwd()
    os.chdir(os.path.dirname(path))      # relative paths/globs resolve here
    try:
        tcl.call("source", path)
    except tkinter.TclError as exc:
        detail = tcl.eval("set ::errorInfo")
        raise ConfigError("Tcl error while reading %s: %s\n%s" % (path, exc, detail))
    finally:
        os.chdir(old_cwd)

    values = {}
    for name in sorted(set(tcl.splitlist(tcl.eval("info globals"))) - before):
        if not IDENT_RE.match(name):
            continue  # flowgen variables are always plain identifiers
        if tcl.eval("array exists ::%s" % name) == "1":
            flat = tcl.splitlist(tcl.eval("array get ::%s" % name))
            values[name] = dict(zip(flat[0::2], flat[1::2]))
        else:
            values[name] = tcl.eval("set ::%s" % name)
    return values, tcl


class FileSet(object):
    """Files collected from a file-list variable (with .f files expanded)."""

    def __init__(self):
        self.files = []
        self.incdirs = []
        self.defines = []
        self.unreadable = False    # a .f file list could not be read


class Config(object):
    """Typed, validated view of the block config. Collects all problems."""

    def __init__(self, path, force=False):
        self.force = force
        self.forced = []      # file errors downgraded to warnings by --force
        self.file_errors = False
        self.path = os.path.realpath(path)   # follow flowgen/configs links to common_files
        self.dir = os.path.dirname(self.path)
        self.errors = []
        self.warnings = []
        self._raw, self._tcl = load_tcl_config(self.path)

        for name in sorted(self._raw):
            if name.startswith(CHECKED_PREFIXES) and name not in KNOWN_VARS:
                self.warnings.append("unknown variable '%s' ignored (typo?)" % name)

        self.top = self.scalar("TOP_DESIGN", required=True)
        self.flows = self.tcl_list("FLOWS", default=["rtl2syn"])
        for flow in self.flows:
            if flow not in SUPPORTED_FLOWS:
                self.errors.append("FLOWS: unknown flow '%s' (supported: %s)"
                                   % (flow, ", ".join(SUPPORTED_FLOWS)))

        # Design data
        self.rtl_format = self.choice("RTL_FORMAT", RTL_READERS, default="sverilog")
        self.rtl = self.fileset("RTL_FILES", "RTL_SEARCH_DIRS", required=True,
                                incdirs=self.paths("RTL_INCDIRS"))
        self.rtl.defines += self.tcl_list("RTL_DEFINES")
        if self.rtl_format == "vhdl" and (self.rtl.incdirs or self.rtl.defines):
            self.errors.append("RTL_INCDIRS / RTL_DEFINES (or +incdir+/+define+ in a .f "
                               "file) only apply to verilog/sverilog")
        needs_syn = "rtl2syn" in self.flows
        self.syn = self.fileset("SYN_FILES", "SYN_SEARCH_DIRS", required=needs_syn, netlist=True)
        self.apr = self.fileset("APR_FILES", netlist=True)        # for future flows
        self.svf = self.single_path("SVF_FILE")
        self.sdc = self.paths("SDC_FILES")                        # not used by Formality
        self.upf = self.paths("UPF_FILES")                        # not used yet

        # Libraries and simulation models
        self.lib_format = self.choice("LIB_FORMAT", LIB_READERS, default="db")
        self.libs = self.paths("LIB_FILES")
        if not self.libs:
            self.file_error("LIB_FILES is empty")
        if self.lib_format in LIB_EXTENSIONS:
            for lib in self.libs:
                if not lib.endswith(LIB_EXTENSIONS[self.lib_format]):
                    self.warnings.append("LIB_FILES: %s does not look like a %s file "
                                         "(LIB_FORMAT is \"%s\")"
                                         % (lib, self.lib_format, self.lib_format))
        self.sim = self.fileset("SIM_FILES", netlist=True)
        self.blackboxes = self.tcl_list("BLACKBOX_MODULES")

        # Tool settings: TOOL_CMD(<tool>), TOOL_SETTINGS(<tool>,<name>), TOOL_HOOKS(<tool>,<point>)
        self.tool_cmd = self.array("TOOL_CMD")
        self.tool_settings = self.keyed_array("TOOL_SETTINGS")
        self.tool_hooks = {}
        for tool, points in self.keyed_array("TOOL_HOOKS").items():
            for point, file_name in points.items():
                if tool == FM and point not in HOOK_POINTS:
                    self.errors.append("TOOL_HOOKS(%s,%s): unknown hook point (valid: %s)"
                                       % (tool, point, ", ".join(HOOK_POINTS)))
                else:
                    self.tool_hooks.setdefault(tool, {})[point] = self.resolve(
                        "TOOL_HOOKS(%s,%s)" % (tool, point), file_name)
        for name in self.tool_settings.get(FM, {}):
            if not IDENT_RE.match(name):
                self.errors.append("TOOL_SETTINGS(%s,%s): not a valid app var name" % (FM, name))

        # Run environment
        self.env_source = self.paths("ENV_SOURCE")
        self.env_setenv = self.array("ENV_SETENV")
        for key in self.env_setenv:
            if not IDENT_RE.match(key):
                self.errors.append("ENV_SETENV(%s): not a valid environment variable name" % key)

    def file_error(self, msg):
        """An error caused by a missing or unreadable input file."""
        if self.force:
            self.forced.append(msg)
            self.warnings.append("%s (ignored: --force)" % msg)
        else:
            self.file_errors = True
            self.errors.append(msg)

    # Typed getters

    def _get(self, name, kind):
        value = self._raw.get(name)
        if value is None:
            return None
        is_array = isinstance(value, dict)
        if is_array != (kind == "array"):
            self.errors.append("%s must be %s" % (name, "an array, e.g. set %s(key) \"value\"" % name
                                                  if kind == "array"
                                                  else "a plain variable, not an array"))
            return None
        return value

    def scalar(self, name, default=None, required=False):
        value = self._get(name, "scalar")
        if value is None or value.strip() == "":
            if required:
                self.errors.append("%s is required" % name)
            return default
        return value.strip()

    def choice(self, name, options, default):
        value = self.scalar(name, default=default)
        if value not in options:
            self.errors.append("%s: \"%s\" is not one of %s"
                               % (name, value, ", ".join(sorted(options))))
        return value

    def tcl_list(self, name, default=None, required=False):
        """List items, with $variables expanded (braces do not expand them in Tcl)."""
        value = self._get(name, "scalar")
        items = []
        for item in (self._tcl.splitlist(value) if value else ()):
            try:
                item = str(self._tcl.call("subst", "-nocommands", "-nobackslashes", item))
            except tkinter.TclError as exc:
                self.errors.append("%s: %s (define the variable in the config, or remove it "
                                   "from the value)" % (name, exc))
                continue
            if item.strip():
                items.append(item.strip())
        if not items:
            if required:
                self.errors.append("%s is required (a list in braces)" % name)
            return list(default or [])
        return items

    def array(self, name):
        return self._get(name, "array") or {}

    def keyed_array(self, name):
        """NAME(<tool>,<key>) -> {tool: {key: value}}."""
        result = {}
        for key, value in self.array(name).items():
            tool, sep, sub = key.partition(",")
            if not sep or not tool or not sub:
                self.errors.append("%s(%s): key must be <tool>,<name>, e.g. %s(%s,...)"
                                   % (name, key, name, FM))
                continue
            result.setdefault(tool.strip(), {})[sub.strip()] = value
        return result

    def resolve(self, label, path, base=None):
        """Absolute path (relative to base, default the config dir); warn if missing."""
        full = os.path.normpath(os.path.join(base or self.dir, os.path.expanduser(path)))
        if not os.path.exists(full):
            self.warnings.append("%s: file not found: %s" % (label, full))
        return full

    def find(self, label, path, search, base=None):
        """Locate a file: absolute as given; relative in the search dirs (in
        order), then relative to base (default the config dir). Warn if missing."""
        path = os.path.expanduser(path)
        base = base or self.dir
        if os.path.isabs(path):
            return self.resolve(label, path)
        dirs = []
        for d in search + [base]:
            if d not in dirs:
                dirs.append(d)
        found = [os.path.normpath(os.path.join(d, path)) for d in dirs
                 if os.path.isfile(os.path.join(d, path))]
        if len(found) > 1:
            self.warnings.append("%s: %s found in several places, using %s"
                                 % (label, path, found[0]))
        if found:
            return found[0]
        self.warnings.append("%s: file not found: %s (searched: %s)"
                             % (label, path, ", ".join(dirs)))
        return os.path.normpath(os.path.join(base, path))

    def expand(self, text):
        """Expand $NAME / ${NAME} in a .f line: config variables first, then
        environment variables. Unknown names are left as they are."""
        def repl(m):
            name = m.group(1) or m.group(2)
            value = self._raw.get(name)
            if isinstance(value, str):
                return value
            return os.environ.get(name, m.group(0))
        return re.sub(r"\$\{(\w+)\}|\$(\w+)", repl, text)

    def paths(self, name, required=False):
        return [self.resolve(name, p) for p in self.tcl_list(name, required=required)]

    def single_path(self, name):
        items = self.paths(name)
        if len(items) > 1:
            self.errors.append("%s: give one file, got %d" % (name, len(items)))
        return items[0] if items else ""

    def fileset(self, name, search_var=None, required=False, netlist=False, incdirs=()):
        """Files from a list variable; entries ending in .f are file lists.

        File names without a full path are looked up in the search_var dirs,
        then the include dirs (incdirs plus +incdir+ from .f files), then
        next to the config or .f file, in that order.
        """
        result = FileSet()
        result.incdirs = list(incdirs)
        search = []
        if search_var:
            search = self.paths(search_var)
            for d in search:
                if os.path.exists(d) and not os.path.isdir(d):
                    self.errors.append("%s: not a directory: %s" % (search_var, d))
        items = self.tcl_list(name)
        if required and not items:
            self.file_error("%s is empty" % name)

        # Collect all names first, so +incdir+ dirs from anywhere in the .f
        # files are known before any file is looked up.
        refs = []                        # (label for messages, name, base dir)
        for item in items:
            if item.endswith(".f"):
                self.read_filelist(name, item, result, refs, [], search)
            else:
                refs.append((name, item, None))
        if netlist and (result.incdirs or result.defines):
            self.warnings.append("%s: +incdir+/+define+ in a .f file ignored for netlists" % name)
            result.incdirs, result.defines = [], []
        lookup = search + [d for d in result.incdirs if d not in search]
        for label, ref, base in refs:
            path = self.find(label, ref, lookup, base)
            if path not in result.files:     # a file listed twice is read once
                result.files.append(path)
        if required and items and not result.files and not result.unreadable:
            self.file_error("%s: the .f file lists contain no files" % name)
        return result

    def read_filelist(self, name, path, result, refs, stack, search, base=None):
        """Read a .f file: one name per line, // or # comments, $VAR expanded
        (config variables, then environment variables).

        Also understands -f/-F <file> (nested lists), -v <file>,
        +incdir+<dir>[+<dir>...] and +define+<MACRO>[=<value>][+...].
        File names go to refs; +incdir+ dirs are relative to the .f file.
        """
        label = "%s (%s)" % (name, os.path.basename(path))
        path = os.path.expanduser(path)
        searched = ""
        if not os.path.isabs(path):
            dirs = search + [base or self.dir]
            found = [os.path.join(d, path) for d in dirs if os.path.isfile(os.path.join(d, path))]
            path = os.path.normpath(found[0] if found else os.path.join(base or self.dir, path))
            searched = " (searched: %s)" % ", ".join(dirs)
        if path in stack:
            self.errors.append("%s: file list includes itself" % label)
            return
        try:
            with open(path) as fh:
                lines = fh.readlines()
        except (IOError, OSError):
            result.unreadable = True
            self.file_error("%s: file list not found or not readable: %s%s"
                            % (label, path, searched))
            return
        base = os.path.dirname(path)
        for lineno, line in enumerate(lines, 1):
            line = line.split("//", 1)[0].strip()
            if not line or line.startswith("#"):
                continue
            tokens = [self.expand(t) for t in line.split()]
            i = 0
            while i < len(tokens):
                tok = tokens[i]
                where = "%s line %d" % (label, lineno)
                if tok in ("-f", "-F", "-v", "-y"):
                    if i + 1 >= len(tokens):
                        self.errors.append("%s: %s needs a file name" % (where, tok))
                        break
                    if tok == "-y":
                        self.warnings.append("%s: -y library directories are not supported, "
                                             "ignored: %s" % (where, tokens[i + 1]))
                    elif tok == "-v":
                        refs.append((where, tokens[i + 1], base))
                    else:
                        self.read_filelist(name, tokens[i + 1], result, refs,
                                           stack + [path], search, base)
                    i += 2
                    continue
                if tok.startswith("+incdir+"):
                    result.incdirs += [self.resolve(where, d, base)
                                       for d in tok[len("+incdir+"):].split("+") if d]
                elif tok.startswith("+define+"):
                    result.defines += [d for d in tok[len("+define+"):].split("+") if d]
                elif tok.startswith(("-", "+")):
                    self.warnings.append("%s: option %s not supported, ignored" % (where, tok))
                else:
                    refs.append((where, tok, base))
                i += 1


# Quoting helpers

_TCL_BARE = re.compile(r"^[A-Za-z0-9_./:+=@%,~-]+$")
_TCL_SPECIAL = re.compile(r'([\\{}\[\]$";# \t])')


def tcl_word(text):
    """Quote text as one Tcl word (also valid as a Tcl list element)."""
    if _TCL_BARE.match(text):
        return text
    if text and not re.search(r"[{}\\]", text):
        return "{%s}" % text
    return _TCL_SPECIAL.sub(r"\\\1", text).replace("\n", "\\n") or "{}"


def tcl_list_block(items, indent="  "):
    """A braced, one-item-per-line Tcl list."""
    if not items:
        return "{}"
    return "{\n%s\n}" % "\n".join(indent + tcl_word(i) for i in items)


def csh_word(label, text):
    """Single-quote a value for csh; reject what csh cannot quote safely."""
    if re.search(r"['!\n]", text):
        raise ConfigError("%s: value may not contain ' or ! or newlines for csh: %r"
                          % (label, text))
    return "'%s'" % text


# Formality script generation

def section(title):
    return ["", "# " + title]


def hook_lines(cfg, point):
    """'source <file>' for TOOL_HOOKS(formality,<point>), or nothing."""
    hook = cfg.tool_hooks.get(FM, {}).get(point)
    if not hook:
        return []
    return ["", "# Hook: %s" % point, "source %s" % tcl_word(hook)]


def vcs_option(fileset):
    """-vcs option carrying include dirs and defines, or ''."""
    opts = ["+incdir+" + d for d in fileset.incdirs] + ["+define+" + d for d in fileset.defines]
    return " -vcs %s" % tcl_word(" ".join(opts)) if opts else ""


def gen_fm_rtl2syn(cfg, flow):
    """Formality script, <run_dir>/<flow>/<flow>.tcl, run from that directory."""
    settings = dict(cfg.tool_settings.get(FM, {}))
    settings.setdefault("synopsys_auto_setup", "true")
    lines = ["# Formality %s: RTL vs synthesized netlist for %s" % (flow, cfg.top)]
    if cfg.forced:
        lines.append("# WARNING: generated with --force, these input problems were ignored:")
        lines += ["#   " + msg for msg in cfg.forced]
    lines += [
        "",
        "set FLOW %s" % tcl_word(flow),
        "set TOP_DESIGN %s" % tcl_word(cfg.top),
        "file delete -force ${FLOW}.status",
    ]

    lines += section("Setup")
    lines += ["set_app_var %s %s" % (n, tcl_word(settings[n])) for n in sorted(settings)]
    lines += hook_lines(cfg, "setup")

    if cfg.svf:
        lines += section("Synthesis guidance")
        lines.append("set_svf %s" % tcl_word(cfg.svf))

    lines += section("Libraries")
    lines += ["%s %s" % (LIB_READERS[cfg.lib_format], tcl_word(lib)) for lib in cfg.libs]

    lines += section("Reference: RTL")
    lines.append("%s -r %s%s" % (RTL_READERS[cfg.rtl_format], tcl_list_block(cfg.rtl.files),
                                 vcs_option(cfg.rtl)))
    if cfg.sim.files:
        lines.append("read_verilog -r %s" % tcl_list_block(cfg.sim.files))

    lines += section("Implementation: synthesized netlist")
    lines.append("read_verilog -i -netlist %s" % tcl_list_block(cfg.syn.files))
    if cfg.sim.files:
        lines.append("read_verilog -i %s" % tcl_list_block(cfg.sim.files))

    if cfg.blackboxes:
        lines += section("Black boxes")
        for module in cfg.blackboxes:
            lines += ["set_black_box r:/WORK/%s" % tcl_word(module),
                      "set_black_box i:/WORK/%s" % tcl_word(module)]

    lines += section("Top-level designs")
    lines += ["set_top r:/WORK/$TOP_DESIGN",
              "set_top i:/WORK/$TOP_DESIGN"]
    lines += hook_lines(cfg, "pre_match")

    lines += section("Match")
    lines += ["match",
              "report_unmatched_points > reports/${FLOW}.unmatched.rpt"]
    lines += hook_lines(cfg, "pre_verify")

    lines += section("Verify")
    lines += [
        "set verify_ok [verify]",
        "if {$verify_ok} {",
        "  set result PASS",
        "} else {",
        "  set result FAIL",
        "  report_failing_points > reports/${FLOW}.failing.rpt",
        "  report_aborted_points > reports/${FLOW}.aborted.rpt",
        "  analyze_points -failing > reports/${FLOW}.analyze.rpt",
        "}",
    ]
    lines += hook_lines(cfg, "post_verify")

    lines += section("Finish")
    lines += [
        "save_session -replace ${FLOW}",
        "set fh [open ${FLOW}.status w]",
        "puts $fh $result",
        "close $fh",
        'puts "$FLOW result: $result"',
        "exit",
        "",
    ]

    with open(path, "w") as fh:
        fh.write(text)
    if executable:

def main(argv=None):
    parser = argparse.ArgumentParser(
        description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter)
    parser.add_argument("config", metavar="CONFIG",
                        help="block config file, typically block.config.tcl")
    parser.add_argument("--force", action="store_true",
                        help="generate even if input files are missing or required file "
                             "lists are empty (those errors become warnings)")
    args = parser.parse_args(argv)

    if not os.path.isfile(args.config):
        sys.exit("flowgen: ERROR: config file not found: %s" % args.config)

    try:
        cfg = Config(args.config, force=args.force)
    except ConfigError as exc:
        sys.exit("flowgen: ERROR: %s" % exc)

    for msg in cfg.warnings:
        print("flowgen: WARNING: %s" % msg)
    if cfg.errors:
        for msg in cfg.errors:
            print("flowgen: ERROR: %s" % msg)
        hint = ""
        if not args.force and cfg.file_errors:
            hint = " Use --force to generate anyway with missing or empty file lists."
        sys.exit("flowgen: %d error(s) in %s; nothing generated.%s"
                 % (len(cfg.errors), cfg.path, hint))

    now = datetime.datetime.now()
    block = re.sub(r"[^A-Za-z0-9_.-]", "_", cfg.top)   # safe as a directory name
    run_dir = os.path.abspath("%s_%s_formality" % (now.strftime("%Y_%m_%d_%H_%M_%S"), block))

        # never leaves a half-written run directory.

    except ConfigError as exc:
        sys.exit("flowgen: ERROR: %s" % exc)

        shutil.rmtree(run_dir)

    shutil.copy2(cfg.path, os.path.join(run_dir, os.path.basename(cfg.path)))

    for flow, (fm_tcl, run_csh) in outputs.items():

        print("flowgen: WARNING: generated with --force; %d input problem(s) ignored "
              "(listed in the script headers)" % len(cfg.forced))
    for flow in cfg.flows:
        print("flowgen:   %-10s run with: %s" % (flow, os.path.join(run_dir, "run_%s.csh" % flow)))
    return 0

    sys.exit(main())

#!/usr/bin/env python3
"""pack - concatenate a repo's files into one plain-text bundle.

Usage:
    pack.py [-o BUNDLE] [ROOT]

ROOT defaults to the git repo containing the current directory (or the
current directory if not in git). In a git repo only tracked files are
packed; otherwise every file under ROOT except .git/ is packed. Paths
under EXCLUDE_DIRS (docs/) are never packed. The repo's common_files/ holds
only the shipped tool files (templates, csv2config.py); the user's own block
configs live only on their machine and are never in git.

The bundle is pure ASCII, so it survives copy/paste, downloads and CRLF
line endings. Pure-ASCII text files are stored as readable lines; any other
file (non-ASCII text, binary) is stored as base64 and restored byte-exact.
Recreate the files with unpack.py (which is itself inside the bundle).

Bundle format:
    #@@PACK 3
    #@@FILE <path> mode=<octal> enc=<text|b64> lines=<n> eol=<0|1> md5=<hex>
    <n lines: file content (text) or base64 of the raw bytes (b64)>
    #@@END <path>

Python 3.6+, standard library only.
"""

import argparse
import base64
import hashlib
import os
import subprocess
import sys

EXCLUDE_DIRS = ("docs/",)   # top-level directories never packed


def repo_root():
    try:
        out = subprocess.check_output(["git", "rev-parse", "--show-toplevel"],
                                      stderr=subprocess.DEVNULL)
        return out.decode().strip()
    except (OSError, subprocess.CalledProcessError):
        return os.getcwd()


def list_files(root):
    """Sorted relative paths with '/' separators."""
    try:
        out = subprocess.check_output(["git", "-C", root, "ls-files", "-z"],
                                      stderr=subprocess.DEVNULL)
        files = [f for f in out.decode("utf-8").split("\0") if f]
    except (OSError, subprocess.CalledProcessError):
        files = []
        for dirpath, dirnames, filenames in os.walk(root):
            dirnames[:] = [d for d in dirnames if d != ".git"]
            for name in filenames:
                rel = os.path.relpath(os.path.join(dirpath, name), root)
                files.append(rel.replace(os.sep, "/"))
    return sorted(f for f in files
                  if os.path.isfile(os.path.join(root, f)) and not f.startswith(EXCLUDE_DIRS))


def main(argv=None):
    parser = argparse.ArgumentParser(description="Pack repo files into one text bundle.")
    parser.add_argument("root", nargs="?", help="directory to pack (default: git repo root)")
    parser.add_argument("-o", "--output", help="bundle file (default: bundle.txt)")
    args = parser.parse_args(argv)

    root = os.path.abspath(args.root or repo_root())
    output = os.path.abspath(args.output or "bundle.txt")

    chunks = ["#@@PACK 3\n"]
    packed, encoded = 0, []
    for rel in list_files(root):
        path = os.path.join(root, rel)
        if os.path.abspath(path) == output:
            continue
        with open(path, "rb") as fh:
            data = fh.read()
        if data.startswith(b"#@@PACK"):
            continue                     # never pack a bundle inside a bundle
        mode = os.stat(path).st_mode & 0o777

        norm = data.replace(b"\r\n", b"\n")
        if all(b in b"\t\n" or 32 <= b < 127 for b in norm):
            enc, data = "text", norm
            text = data.decode("ascii")
            eol = 1 if text.endswith("\n") or not text else 0
            lines = text.split("\n")
            if eol:
                lines = lines[:-1]       # drop the empty piece after the last newline
        else:
            enc, eol = "b64", 1
            b64 = base64.b64encode(data).decode("ascii")
            lines = [b64[i:i + 76] for i in range(0, len(b64), 76)]
            encoded.append(rel)
        md5 = hashlib.md5(data).hexdigest()

        chunks.append("#@@FILE %s mode=%03o enc=%s lines=%d eol=%d md5=%s\n"
                      % (rel, mode, enc, len(lines), eol, md5))
        chunks.extend(line + "\n" for line in lines)
        chunks.append("#@@END %s\n" % rel)
        packed += 1

    with open(output, "w", encoding="ascii", newline="\n") as fh:
        fh.writelines(chunks)

    for rel in encoded:
        print("pack: note: not pure ASCII text, stored as base64: %s" % rel)
    print("pack: %d files from %s -> %s" % (packed, root, output))
    return 0


if __name__ == "__main__":
    sys.exit(main())
#@@END pack.py

## pack.py / unpack.py

- `python3 pack.py [-o BUNDLE] [ROOT]`, default output bundle.txt.
- Packs only git-tracked files (so `git add` new files BEFORE packing).
  Excludes docs/ (EXCLUDE_DIRS). The repo's common_files/ holds only the
  shipped tool files; user configs are never in git.
- Bundle format 3 (`#@@PACK 3`), pure ASCII. Each file has a header
  `#@@FILE <path> mode=<octal> enc=<text|b64> lines=<n> eol=<0|1> md5=<hex>`,
  its lines, then `#@@END <path>`. Pure-ASCII text files are stored as
  readable lines; anything else as base64. This survives charset
  conversion and CRLF line endings (a non-ASCII bundle was once corrupted
  in transit: byte 0xbf at offset 114).
- `python3 unpack.py BUNDLE [-d DIR] [--force] [--list]`
  - Validates the whole bundle (MD5 per file) before writing anything.
  - Owned dirs flowgen/ and examples/: error if they exist, unless
    --force, which removes and recreates them.
  - Shared dir common_files/: bundle files (templates, csv2config.py,
    README) are written/overwritten; nothing else there is removed or
    changed. User configs are never touched.
  - Top-level files (pack.py, unpack.py, session.md, .gitignore) are
    overwritten.
  - Then links every DIR/common_files/*.config.tcl into flowgen/configs/.
    Warns if common_files/ is missing.
  - Only `--force` (double dash) is accepted.
- Regenerate the bundle after every change and commit it.

## flowgen.py

Usage:
```
runflowgen.sh [--force] CONFIG
flowgen.py [--force] CONFIG
```
- CONFIG: block config, typically block.config.tcl (any name works),
  usually given as configs/block.config.tcl.
- --force: missing or unreadable input files (e.g. a missing .f list)
  and empty required file lists (RTL_FILES, SYN_FILES, LIB_FILES)
  become warnings and the run directory is generated anyway; the ignored
  problems are listed in the generated script header. Other config
  errors (missing settings, bad values, Tcl errors) still stop it.

Output (in the current directory; overwritten if it already exists):
```
<yyyy_mm_dd_hh_mm_ss>_<TOP_DESIGN>_formality/
  block.config.tcl      copy of the config used (original file name)
  run_rtl2syn.csh       tcsh: source ENV_SOURCE, setenv ENV_SETENV, run tool
  rtl2syn/              the tool runs here
    rtl2syn.tcl         generated Formality script
    logs/               empty; log goes to logs/rtl2syn.log
    reports/            empty; reports go here
```
- run_rtl2syn.csh runs `<TOOL_CMD> -f rtl2syn.tcl < /dev/null |& tee
  logs/rtl2syn.log` from rtl2syn/, then reads rtl2syn/rtl2syn.status.
  Exit codes: 0 PASS, 1 FAIL, 2 did not finish.
- Generated scripts are kept simple: one-line header, no generation date
  or "generated by" comments, no placeholder comments, 2-space indent.

How it works:
- The config is sourced in a real Tcl interpreter (tkinter.Tcl()) with
  the working directory set to the config's real directory. Only global
  variables created by the config are read; arrays become dicts.
- Config path is resolved with realpath, so relative paths are relative
  to common_files/ even through a flowgen/configs/ link.
- `$VARIABLES` inside braced list items are expanded by flowgen
  (Tcl `subst -nocommands -nobackslashes`).
- All values are validated; all errors are reported together; nothing is
  written unless everything renders. Unknown names with known prefixes
  (DESIGN, TOP_, RTL_, SYN_, LIB_, TOOL_, ENV_, ...) produce a typo
  warning. Missing files are warnings.
- .f file lists are expanded by flowgen: one path per line, // and #
  comments, $VAR from the environment, relative to the .f file. Supports
  nested -f/-F, -v, +incdir+, +define+. -y and other options are ignored
  with a warning.
- Relative file names in RTL_FILES/SYN_FILES and their .f lists (and .f
  files themselves) are looked up in RTL_SEARCH_DIRS/SYN_SEARCH_DIRS in
  order, then (RTL only) RTL_INCDIRS and +incdir+ dirs, then next to the
  config or .f file. All .f files are read first, so +incdir+ lines apply
  to every file name.
- $NAME / ${NAME} in .f files: config variables first (e.g. USER_RTL_DIR),
  then environment variables; unknown names are left as is. First match wins (warning if
  found in several). Duplicate files are read once.

DIR/common_files/, the template included:
    flowgen/configs/<name> -> ../../common_files/<name>
Re-run unpack (with --force) to pick up configs added to common_files/ later.

If a top-level directory from the bundle (flowgen/, examples/) already
exists in DIR, unpack stops with an error and changes nothing. With --force
those directories are removed first and recreated from the bundle, so no
stale files are left behind.

common_files/ is shared with you: the bundle's files there (the templates
and csv2config.py) are written or overwritten, and nothing else in it is
removed or changed. Your own block configs are never touched. Top-level
files (pack.py, unpack.py, ...) are overwritten.

Every file is checked against its MD5 before anything is written; a damaged
bundle writes nothing.

Python 3.6+, standard library only.
"""

import argparse
import base64
import binascii
import hashlib
import os
import re
import shutil
import sys

CONFIG_SRC = "common_files"          # your golden configs + shipped templates
SHARED_DIRS = (CONFIG_SRC,)           # bundle files written here, nothing removed
CONFIG_LINK_DIR = "flowgen/configs"   # created by unpack, one link per config

HEADER_RE = re.compile(r"^#@@FILE (\S+) mode=([0-7]{3}) enc=(text|b64) lines=(\d+) "
                       r"eol=([01]) md5=([0-9a-f]{32})$")


class BundleError(Exception):
    pass


def read_bundle(bundle_path):
    with open(bundle_path, "rb") as fh:
        raw = fh.read()
    if raw.startswith(b"\xef\xbb\xbf"):
        raw = raw[3:]                   # BOM added by some editors
    try:
        return raw.decode("ascii")
    except UnicodeDecodeError as exc:
        raise BundleError("bundle has a non-ASCII byte 0x%02x at offset %d: it was changed "
                          "during transfer (character-set conversion). Bundles are pure "
                          "ASCII; copy it again as-is (e.g. download the raw file)."
                          % (raw[exc.start], exc.start))


def parse(bundle_path):
    """Return [(path, mode, bytes)] after validating the whole bundle."""
    lines = read_bundle(bundle_path).splitlines()
    if not lines or lines[0] != "#@@PACK 3":
        raise BundleError("not a pack.py bundle, or made by a different pack.py version "
                          "(first line must be '#@@PACK 3')")

    files, i = [], 1
    while i < len(lines):
        if lines[i].strip() == "":
            i += 1                      # tolerate blank lines between files
            continue
        m = HEADER_RE.match(lines[i].rstrip())
        if not m:
            raise BundleError("line %d: expected #@@FILE header, got: %r" % (i + 1, lines[i][:80]))
        path, mode, enc, count, eol, md5 = m.groups()
        count = int(count)
        check_path(path, i)

        body = lines[i + 1:i + 1 + count]
        end = i + 1 + count
        if len(body) != count or end >= len(lines) or lines[end].rstrip() != "#@@END " + path:
            raise BundleError("%s: truncated or damaged (missing #@@END)" % path)

        if enc == "text":
            data = ("\n".join(body) + ("\n" if eol == "1" and count else "")).encode("ascii")
        else:
            try:
                data = base64.b64decode("".join(l.strip() for l in body), validate=True)
            except (binascii.Error, ValueError):
                raise BundleError("%s: damaged base64 data" % path)
        if hashlib.md5(data).hexdigest() != md5:
            raise BundleError("%s: MD5 mismatch (content was changed in transit)" % path)
        files.append((path, int(mode, 8), data))
        i = end + 1
    return files


def check_path(path, i):
    if path.startswith("/") or ".." in path.split("/"):
        raise BundleError("line %d: unsafe path: %s" % (i + 1, path))


def link_configs(out_dir):
    """Create flowgen/configs/ with a relative link to each golden config."""
    src = os.path.join(out_dir, CONFIG_SRC)
    link_dir = os.path.join(out_dir, *CONFIG_LINK_DIR.split("/"))
    os.makedirs(link_dir, exist_ok=True)
    if not os.path.isdir(src):
        print("unpack: WARNING: %s not found; %s left empty"
              % (os.path.abspath(src), CONFIG_LINK_DIR))
        return
    rel_src = os.path.relpath(src, link_dir)       # ../../common_files
    names = sorted(n for n in os.listdir(src)
                   if n.endswith(".config.tcl") and os.path.isfile(os.path.join(src, n)))
    linked = 0
    for name in names:
        dest = os.path.join(link_dir, name)
        if os.path.lexists(dest):        # a file of that name came with the bundle
            print("unpack: WARNING: %s/%s already exists; not linked" % (CONFIG_LINK_DIR, name))
            continue
        os.symlink(os.path.join(rel_src, name), dest)
        linked += 1
    print("unpack: linked %d config(s) from %s into %s" % (linked, CONFIG_SRC, CONFIG_LINK_DIR))


def main(argv=None):
    parser = argparse.ArgumentParser(description="Recreate files from a pack.py bundle.")
    parser.add_argument("bundle", help="bundle file made by pack.py")
    parser.add_argument("-d", "--dir", default=".", help="output directory (default: .)")
    parser.add_argument("--force", action="store_true",
                        help="remove the existing flowgen/ and examples/ before unpacking; "
                             "common_files/ and everything else is kept")
    parser.add_argument("--list", action="store_true", help="only list the files")
    args = parser.parse_args(argv)

    try:
        files = parse(args.bundle)
    except (OSError, BundleError) as exc:
        sys.exit("unpack: ERROR: %s" % exc)

    if args.list:
        for path, mode, _ in files:
            print("%03o %s" % (mode, path))
        return 0

    # Top-level directories (flowgen) this bundle owns.
    tops = sorted(set(p.split("/")[0] for p, _, _ in files
                      if "/" in p and p.split("/")[0] not in SHARED_DIRS))
    existing = [t for t in tops if os.path.lexists(os.path.join(args.dir, t))]
    if existing and not args.force:
        sys.exit("unpack: ERROR: already exists in %s: %s\n"
                 "unpack: use --force to remove and recreate them."
                 % (os.path.abspath(args.dir), ", ".join(existing)))
    for top in existing:
        target = os.path.join(args.dir, top)
        print("unpack: removing existing %s" % target)
        if os.path.isdir(target) and not os.path.islink(target):
            shutil.rmtree(target)
        else:
            os.remove(target)

    for path, mode, data in files:
        dest = os.path.join(args.dir, *path.split("/"))
        if os.path.dirname(dest):
            os.makedirs(os.path.dirname(dest), exist_ok=True)
        with open(dest, "wb") as fh:
            fh.write(data)
        os.chmod(dest, mode)

    link_configs(args.dir)
    print("unpack: %d files -> %s" % (len(files), os.path.abspath(args.dir)))
    return 0


if __name__ == "__main__":
    sys.exit(main())
